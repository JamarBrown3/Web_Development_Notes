What Is React

React builds a page out of components. A component is a function that returns what should be on the screen. When its state changes, React runs the function again and updates the page. Terminal setup is on the Setup page. Every example here uses items, so use Find and Replace to change items to games, posts, or whatever your project needs.

A whole form in one object

One handleChange works for every field, as long as each input's name matches a key in the object. Always copy the old object with ...previous, or the other fields get lost.

import { useState } from "react";

const SignUp = () => {
  const [form, setForm] = useState({ name: "", email: "" });

  const handleChange = (event) => {
    const { name, value } = event.target;

    setForm((previous) => ({ ...previous, [name]: value }));
  };

  const handleSubmit = (event) => {
    event.preventDefault(); // stops the browser from reloading the page
    console.log(form);
  };

  return (
    <form onSubmit={handleSubmit}>
      <label htmlFor="name">Name</label>
      <input id="name" name="name" value={form.name} onChange={handleChange} />

      <label htmlFor="email">Email</label>
      <input id="email" name="email" type="email" value={form.email} onChange={handleChange} />

      <button type="submit">Submit</button>
    </form>
  );
};

export default SignUp;

Add, update, and remove in an array

Never use push or splice on state. Make a new array so React can see that it changed.

import { useState } from "react";

const Todos = () => {
  const [todos, setTodos] = useState([]);
  const [text, setText] = useState("");

  const addTodo = (event) => {
    event.preventDefault();
    if (!text.trim()) return;

    setTodos((previous) => [
      ...previous,
      { id: crypto.randomUUID(), text: text.trim(), done: false },
    ]);
    setText("");
  };

  const toggleTodo = (id) => {
    setTodos((previous) =>
      previous.map((todo) => (todo.id === id ? { ...todo, done: !todo.done } : todo)),
    );
  };

  const removeTodo = (id) => {
    setTodos((previous) => previous.filter((todo) => todo.id !== id));
  };

  return (
    <div>
      <form onSubmit={addTodo}>
        <input value={text} onChange={(event) => setText(event.target.value)} />
        <button type="submit">Add</button>
      </form>

      <ul>
        {todos.map((todo) => (
          <li key={todo.id}>
            <label>
              <input type="checkbox" checked={todo.done} onChange={() => toggleTodo(todo.id)} />
              {todo.text}
            </label>
            <button type="button" onClick={() => removeTodo(todo.id)}>Remove</button>
          </li>
        ))}
      </ul>
    </div>
  );
};

export default Todos;

useEffect

useEffect runs code after the component is on the screen. Use it to reach outside of React, such as fetching data, starting a timer, or listening for events. The array at the end says when to run it again.

The three dependency options

import { useEffect } from "react";

// Runs once, when the component first shows up.
useEffect(() => {
  console.log("Page loaded");
}, []);

// Runs once, and again every time id changes.
useEffect(() => {
  console.log("New id:", id);
}, [id]);

// No array: runs after every redraw. You almost never want this.
useEffect(() => {
  console.log("Redrew");
});

Fetch data when the page loads

This is the pattern I use the most. It tracks loading and error, and the ignore flag drops a response that comes back after the user has already left the page. Keep the ignore flag, because React StrictMode runs effects twice in development.

import { useEffect, useState } from "react";

const Items = () => {
  const [items, setItems] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");

  useEffect(() => {
    let ignore = false;

    const fetchItems = async () => {
      try {
        const response = await fetch("/api/items");

        if (!response.ok) {
          throw new Error(`Request failed (${response.status}).`);
        }

        const data = await response.json();

        if (!ignore) {
          setItems(data);
        }
      } catch (problem) {
        if (!ignore) {
          console.error("Error loading items:", problem);
          setError("Unable to load items. Please try again.");
        }
      } finally {
        if (!ignore) {
          setLoading(false);
        }
      }
    };

    fetchItems();

    return () => {
      ignore = true;
    };
  }, []);

  if (loading) return <p role="status">Loading...</p>;
  if (error) return <p role="alert">{error}</p>;

  return (
    <ul>
      {items.map((item) => (
        <li key={item.id}>{item.title}</li>
      ))}
    </ul>
  );
};

export default Items;

A timer with cleanup

This is Countdown.jsx from xenoblade1_Locations. The function returned from useEffect is the cleanup. It stops the timer when the component goes away.

import { useEffect, useState } from "react";

const getTimeLeft = (startsAt) => {
  const diff = new Date(startsAt).getTime() - Date.now();
  if (diff <= 0) return null;
  return {
    days: Math.floor(diff / (1000 * 60 * 60 * 24)),
    hours: Math.floor((diff / (1000 * 60 * 60)) % 24),
    minutes: Math.floor((diff / (1000 * 60)) % 60),
    seconds: Math.floor((diff / 1000) % 60),
  };
};

const Countdown = ({ startsAt }) => {
  const [timeLeft, setTimeLeft] = useState(() => getTimeLeft(startsAt));

  useEffect(() => {
    const timer = setInterval(() => setTimeLeft(getTimeLeft(startsAt)), 1000);
    return () => clearInterval(timer);
  }, [startsAt]);

  if (!timeLeft) {
    return <p>This event has passed.</p>;
  }

  return (
    <p>
      {timeLeft.days}d {timeLeft.hours}h {timeLeft.minutes}m {timeLeft.seconds}s
    </p>
  );
};

export default Countdown;

useRef

useRef keeps a value that survives redraws, but changing it does not redraw the page. Use it for a value the screen does not show, or to reach a real HTML element.

Stop a double click from submitting twice

State updates wait for the next redraw, so two fast clicks could both get through. A ref changes right away.

import { useRef, useState } from "react";

const SaveButton = ({ onSave }) => {
  const [saving, setSaving] = useState(false);
  const inFlight = useRef(false);

  const handleClick = async () => {
    if (inFlight.current) return;

    inFlight.current = true;
    setSaving(true);

    try {
      await onSave();
    } finally {
      inFlight.current = false;
      setSaving(false);
    }
  };

  return (
    <button type="button" disabled={saving} onClick={handleClick}>
      {saving ? "Saving..." : "Save"}
    </button>
  );
};

export default SaveButton;

useContext

Context shares a value with every component inside the Provider, so you do not have to pass props down through each layer. The login on the Security page uses this same pattern.

client/src/context/CartContext.jsx

import { createContext, useContext, useState } from "react";

const CartContext = createContext(null);

export function CartProvider({ children }) {
  const [cart, setCart] = useState([]);

  const addToCart = (item) => {
    setCart((previous) => [...previous, item]);
  };

  const value = { cart, addToCart, count: cart.length };

  return <CartContext.Provider value={value}>{children}</CartContext.Provider>;
}

export function useCart() {
  return useContext(CartContext);
}

Custom Hooks

A custom hook is a function that starts with use and calls other hooks. It lets several components share the same logic.

client/src/hooks/useFetch.js

This is the fetch pattern from above, packed into one line for your pages to use. Write const { data, loading, error } = useFetch("/api/items");

import { useEffect, useState } from "react";

export default function useFetch(url) {
  const [data, setData] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");

  useEffect(() => {
    let ignore = false;

    const load = async () => {
      setLoading(true);
      setError("");

      try {
        const response = await fetch(url);

        if (!response.ok) {
          throw new Error(`Request failed (${response.status}).`);
        }

        const json = await response.json();

        if (!ignore) setData(json);
      } catch (problem) {
        if (!ignore) setError(problem.message);
      } finally {
        if (!ignore) setLoading(false);
      }
    };

    load();

    return () => {
      ignore = true;
    };
  }, [url]);

  return { data, loading, error };
}

client/src/hooks/useElementWidth.js

Measures an element and updates when it resizes. Rev uses this to redraw charts at the right size. Write const [ref, width] = useElementWidth(); and put ref={ref} on the element.

import { useEffect, useRef, useState } from "react";

// Measures an element's width and updates when it resizes (charts use this to
// redraw at the right size). Returns [ref, width].
export default function useElementWidth(initial = 600) {
  const ref = useRef(null);
  const [width, setWidth] = useState(initial);

  useEffect(() => {
    const element = ref.current;
    if (!element) return undefined;
    setWidth(element.clientWidth || initial);
    const observer = new ResizeObserver(([entry]) => {
      setWidth(Math.round(entry.contentRect.width));
    });
    observer.observe(element);
    return () => observer.disconnect();
  }, [initial]);

  return [ref, width];
}

client/src/components/DeleteButton.jsx

A delete button with an inline "are you sure" box instead of a browser pop-up. It also shows the error if the delete fails. onDelete needs to be an async function that throws on failure.

import { useId, useRef, useState } from "react";

export default function DeleteButton({ itemName, onDelete, label = "Delete", disabled = false }) {
  const [confirming, setConfirming] = useState(false);
  const [pending, setPending] = useState(false);
  const [error, setError] = useState("");
  const inFlight = useRef(false); // blocks a double click from deleting twice
  const trigger = useRef(null);
  const messageId = useId();

  function cancel() {
    if (inFlight.current) return;
    setConfirming(false);
    setError("");
    trigger.current?.focus();
  }

  async function confirm() {
    if (inFlight.current || disabled) return;
    inFlight.current = true;
    setPending(true);
    setError("");
    try {
      await onDelete();
      setConfirming(false);
    } catch (problem) {
      setError(problem.message || "Couldn't delete it. Try again.");
    } finally {
      inFlight.current = false;
      setPending(false);
    }
  }

  return (
    <div className="delete-control">
      <button
        ref={trigger}
        type="button"
        disabled={disabled || pending}
        aria-expanded={confirming}
        onClick={() => setConfirming(true)}
      >
        {label}
      </button>

      {confirming && (
        <div
          role="group"
          aria-labelledby={messageId}
          onKeyDown={(event) => {
            if (event.key === "Escape") {
              event.preventDefault();
              cancel();
            }
          }}
        >
          <p id={messageId}>Delete "{itemName}"? This can't be undone.</p>
          {error && <p role="alert">{error}</p>}
          <button type="button" autoFocus disabled={pending} onClick={cancel}>
            Keep it
          </button>
          <button type="button" disabled={pending} onClick={confirm}>
            {pending ? "Deleting..." : "Yes, delete"}
          </button>
        </div>
      )}
    </div>
  );
}

client/src/theme.js for light and dark mode

This came from Rev. The choice is saved in localStorage. Call applySavedTheme() in main.jsx before render, then write your colors under html[data-theme="dark"] in your CSS.

const KEY = "app-theme";

function paint(theme) {
  document.documentElement.dataset.theme = theme;
}

export function applySavedTheme() {
  let saved = null;
  try {
    saved = localStorage.getItem(KEY);
  } catch {
    // Private windows can block storage; the default theme still works.
  }
  paint(saved === "dark" ? "dark" : "light");
}

export function currentTheme() {
  return document.documentElement.dataset.theme === "dark" ? "dark" : "light";
}

export function saveTheme(theme) {
  paint(theme);
  try {
    localStorage.setItem(KEY, theme);
  } catch {
    // Ignore: the theme still changes for this visit.
  }
}

Talking To The Server

client/src/services/ItemsAPI.js

All five requests live in one file so pages stay short. readResponse throws the server's error message, and the page shows it. This works with both the Postgres and MongoDB examples.

const readResponse = async (response) => {
  const data = await response.json().catch(() => null);

  if (!response.ok) {
    throw new Error(data?.error || `Request failed (${response.status}).`);
  }

  if (data === null) {
    throw new Error("The server returned an unexpected response.");
  }

  return data;
};

const getAllItems = async () => {
  const response = await fetch("/api/items");
  return readResponse(response);
};

const getItemById = async (id) => {
  const response = await fetch(`/api/items/${encodeURIComponent(id)}`);
  return readResponse(response);
};

const createItem = async (item) => {
  const response = await fetch("/api/items", {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
    },
    body: JSON.stringify(item),
  });
  return readResponse(response);
};

    },
    body: JSON.stringify(item),
  });

  });

import { useEffect, useState } from "react";
import ItemsAPI from "../services/ItemsAPI";
import Card from "../components/Card";
import DeleteButton from "../components/DeleteButton";

const Items = () => {
  const [items, setItems] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");

  useEffect(() => {
    let ignore = false;

    const fetchItems = async () => {
      try {
        const data = await ItemsAPI.getAllItems();

        if (!ignore) setItems(data);
      } catch (problem) {
        if (!ignore) setError(problem.message);
      } finally {
        if (!ignore) setLoading(false);
      }
    };

    fetchItems();

    return () => {
      ignore = true;
    };
  }, []);

  const handleDelete = async (item) => {
    await ItemsAPI.deleteItem(item.id);
    setItems((previous) => previous.filter((other) => other.id !== item.id));
  };

  if (loading) return <p role="status">Loading...</p>;
  if (error) return <p role="alert">{error}</p>;

  return (
    <section>
      {items.length === 0 && <p>Nothing saved yet. Add your first one!</p>}

      {items.map((item) => (
        <div key={item.id}>
          <Card
            id={item.id}
            title={item.title}
            image={item.image}
            description={item.description}
          />
          <DeleteButton itemName={item.title} onDelete={() => handleDelete(item)} />
        </div>
      ))}
    </section>
  );
};

client/src/components/ItemForm.jsx

The create page and the edit page share this one form. The page passes in what to do with the saved values through onSubmit.

import { useRef, useState } from "react";

const defaults = { title: "", description: "", image: "" };

export default function ItemForm({ initialValues = defaults, heading, submitLabel, onSubmit }) {
  const [item, setItem] = useState({ ...defaults, ...initialValues });
  const [error, setError] = useState("");
  const [saving, setSaving] = useState(false);
  const submitting = useRef(false);

  const handleChange = ({ target: { name, value } }) => {
    setItem((previous) => ({ ...previous, [name]: value }));
    setError("");
  };

  const handleSubmit = async (event) => {
    event.preventDefault();
    if (submitting.current) return;

    if (!item.title.trim()) {
      setError("Enter a title.");
      return;
    }

    submitting.current = true;
    setSaving(true);

    try {
      await onSubmit({
        title: item.title.trim(),
        description: item.description.trim(),
        image: item.image.trim(),
      });
    } catch (problem) {
      setError(problem.message || "Unable to save. Please try again.");
    } finally {
      submitting.current = false;
      setSaving(false);
    }
  };

  return (
    <section>
      <h2>{heading}</h2>

      <form onSubmit={handleSubmit}>
        <fieldset disabled={saving}>
          <label htmlFor="title">Title</label>
          <input id="title" name="title" value={item.title} onChange={handleChange} maxLength={255} required />

          <label htmlFor="description">Description</label>
          <textarea id="description" name="description" rows="5" value={item.description} onChange={handleChange} />

          <label htmlFor="image">Image URL</label>
          <input id="image" name="image" value={item.image} onChange={handleChange} />

          {error && <p role="alert">{error}</p>}

          <button type="submit">{saving ? "Saving..." : submitLabel}</button>
        </fieldset>
      </form>
    </section>
  );
}

The form only mounts after the item loads, so its inputs start with the saved values. The key makes React start a fresh form when the id changes.

import { useEffect, useState } from "react";
import { useNavigate, useParams } from "react-router-dom";
import ItemForm from "../components/ItemForm";
import ItemsAPI from "../services/ItemsAPI";

export default function EditItem() {
  const { id } = useParams();
  const navigate = useNavigate();
  const [result, setResult] = useState(null);

  useEffect(() => {
    let ignore = false;

    const load = async () => {
      try {
        const item = await ItemsAPI.getItemById(id);
        if (!ignore) setResult({ id, item });
      } catch (problem) {
        if (!ignore) setResult({ id, error: problem.message });
      }
    };

    load();

    return () => {
      ignore = true;
    };
  }, [id]);

  // Never show the previous record while a different id is loading.
  if (result?.id !== id) return <p role="status">Loading...</p>;
  if (result.error) return <p role="alert">{result.error}</p>;

  const updateItem = async (item) => {
    await ItemsAPI.updateItem(id, item);
    navigate("/");
  };

  return (
    <ItemForm
      key={id}
      initialValues={result.item}
      heading="Edit Item"
      submitLabel="Save Changes"
      onSubmit={updateItem}
    />
  );
}
