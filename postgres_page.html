What Is Postgres

Postgres stores data in tables with rows and columns, and every column has a set type. You talk to it with SQL. Tables connect to each other through ids. The .env file for Postgres is on the Setup page.

Project Setup

Install the package

Run this from the folder that has your server package.json.

npm install pg

Local Postgres on a Mac

Use this when you want a database on your computer instead of a hosted one. A Homebrew install uses your Mac username with no password.

brew services start postgresql@14
createdb my_project
psql my_project

server/config/database.js

The pool keeps a few open connections and hands one to each query. Hosted databases like Render need SSL, and local ones do not. rejectUnauthorized: false still encrypts the connection, but it skips checking the host's certificate, so this is a class project setting.

import "dotenv/config";
import pg from "pg";

const isLocal = ["localhost", "127.0.0.1"].includes(process.env.PGHOST);

const config = {
  user: process.env.PGUSER,
  password: process.env.PGPASSWORD,
  host: process.env.PGHOST,
  port: process.env.PGPORT,
  database: process.env.PGDATABASE,
  ssl: isLocal ? false : { rejectUnauthorized: false },
};

export const pool = new pg.Pool(config);

server/config/reset.js

import "dotenv/config";
import { pool } from "./database.js";
import itemData from "../data/items.js";

const createTables = async () => {
  const createTableQuery = `
    DROP TABLE IF EXISTS items;

    CREATE TABLE items (
      id SERIAL PRIMARY KEY,
      title VARCHAR(255) NOT NULL,
      description TEXT NOT NULL DEFAULT '',
      image TEXT NOT NULL DEFAULT '',
      created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
    );
  `;

  await pool.query(createTableQuery);
  console.log("🎉 Items table created.");
};

const seedTables = async () => {
  const insertQuery = `
    INSERT INTO items (title, description, image)
    VALUES ($1, $2, $3);
  `;

  for (const item of itemData) {
    const values = [item.title, item.description, item.image];

    await pool.query(insertQuery, values);
    console.log(`✅ ${item.title} added successfully`);
  }
};

try {
  await createTables();
  await seedTables();
  console.log(`🎉 Finished! Saved ${itemData.length} items.`);
} catch (error) {
  console.error("⚠️ Reset failed:", error.message);
  process.exitCode = 1;
} finally {
  await pool.end();
}

Full CRUD Controller

server/controllers/items.js

The $1 placeholders keep user input separate from the SQL, which is what stops SQL injection. RETURNING * sends back the saved row. Each route also checks the id first and answers 404 when no row matched. The update replaces every field, so send them all, which the edit form on the React page does. This uses the routes file from the Node page.

import { pool } from "../config/database.js";

// Validate IDs before querying PostgreSQL.
const readId = (value) => {
  if (!/^[0-9]+$/.test(String(value))) return null;

  const id = Number(value);

  return Number.isInteger(id) && id > 0 && id <= 2147483647 ? id : null;
};

// Return an error message, or an empty string when everything is valid.
const validateItem = (item) => {
  if (!item || typeof item !== "object" || Array.isArray(item)) {
    return "Item information is required.";
  }

  if (
    typeof item.title !== "string" ||
    !item.title.trim() ||
    item.title.trim().length > 255
  ) {
    return "Enter a title between 1 and 255 characters.";
  }

  return "";
};

// Match the order of the SQL placeholders.
const itemValues = (item) => [
  item.title.trim(),
  String(item.description ?? "").trim(),
  String(item.image ?? "").trim(),
];

// Log the real error, but never send database details to the browser.
const serverError = (res, error) => {
  console.error("Items database error:", error);

  return res.status(500).json({
    error: "Unable to complete the request. Please try again.",
  });
};

const getItems = async (req, res) => {
  try {
    const results = await pool.query("SELECT * FROM items ORDER BY id ASC");

    return res.status(200).json(results.rows);
  } catch (error) {
    return serverError(res, error);
  }
};

const getItemById = async (req, res) => {
  const id = readId(req.params.id);

  if (id === null) {
    return res.status(400).json({ error: "Invalid item ID." });
  }

  try {
    const results = await pool.query("SELECT * FROM items WHERE id = $1", [id]);

    if (results.rows.length === 0) {
      return res.status(404).json({ error: "Item not found." });
    }

    return res.status(200).json(results.rows[0]);
  } catch (error) {
    return serverError(res, error);
  }
};

const createItem = async (req, res) => {
  const validationError = validateItem(req.body);

  if (validationError) {
    return res.status(400).json({ error: validationError });
  }

  try {
    const results = await pool.query(
      `INSERT INTO items (title, description, image)
      VALUES ($1, $2, $3)
      RETURNING *`,
      itemValues(req.body),
    );

    return res.status(201).json(results.rows[0]);
  } catch (error) {
    return serverError(res, error);
  }
};

const updateItem = async (req, res) => {
  const id = readId(req.params.id);

  if (id === null) {
    return res.status(400).json({ error: "Invalid item ID." });
  }

  const validationError = validateItem(req.body);

  if (validationError) {
    return res.status(400).json({ error: validationError });
  }

  try {
    const results = await pool.query(
      `UPDATE items SET
        title = $1,
        description = $2,
        image = $3
      WHERE id = $4
      RETURNING *`,
      [...itemValues(req.body), id],
    );

    if (results.rows.length === 0) {
      return res.status(404).json({ error: "Item not found." });
    }

    return res.status(200).json(results.rows[0]);
  } catch (error) {
    return serverError(res, error);
  }
};

const deleteItem = async (req, res) => {
  const id = readId(req.params.id);

  if (id === null) {
    return res.status(400).json({ error: "Invalid item ID." });
  }

  try {
    const results = await pool.query(
      "DELETE FROM items WHERE id = $1 RETURNING *",
      [id],
    );

    if (results.rows.length === 0) {
      return res.status(404).json({ error: "Item not found." });
    }

    return res.status(200).json(results.rows[0]);
  } catch (error) {
    return serverError(res, error);
  }
};

export default {
  getItems,
  getItemById,
  createItem,
  updateItem,
  deleteItem,
};

server/config/reset.js with a transaction

BEGIN and COMMIT make the whole reset count as one step. If anything fails, ROLLBACK puts the database back the way it was. A transaction has to use one client from the pool, not pool.query.

import "dotenv/config";
import { pool } from "./database.js";
import locationData from "../data/locations.js";
import eventData from "../data/events.js";

const createTables = async (client) => {
  const createTableQuery = `
    DROP TABLE IF EXISTS events;
    DROP TABLE IF EXISTS locations;

    CREATE TABLE locations (
      id SERIAL PRIMARY KEY,
      name VARCHAR(255) NOT NULL UNIQUE,
      description TEXT NOT NULL,
      image TEXT NOT NULL
    );

    CREATE TABLE events (
      id SERIAL PRIMARY KEY,
      location_id INTEGER NOT NULL REFERENCES locations(id) ON DELETE CASCADE,
      title VARCHAR(255) NOT NULL,
      description TEXT NOT NULL,
      starts_at TIMESTAMPTZ NOT NULL
    );
  `;

  await client.query(createTableQuery);
  console.log("🎉 Locations and events tables created.");
};

const seedTables = async (client) => {
  // Save each new location's id so the events can point at it.
  const locationIds = new Map();

  for (const location of locationData) {
    const results = await client.query(
      `INSERT INTO locations (name, description, image)
      VALUES ($1, $2, $3)
      RETURNING id;`,
      [location.name, location.description, location.image],
    );

    locationIds.set(location.name, results.rows[0].id);
  }

  for (const event of eventData) {
    const locationId = locationIds.get(event.locationName);

    if (locationId === undefined) {
      throw new Error(`Unknown location "${event.locationName}" for event "${event.title}".`);
    }

    await client.query(
      `INSERT INTO events (location_id, title, description, starts_at)
      VALUES ($1, $2, $3, $4);`,
      [locationId, event.title, event.description, event.starts_at],
    );
  }
};

const resetDatabase = async () => {
  const client = await pool.connect();

  try {
    await client.query("BEGIN");

    await createTables(client);
    await seedTables(client);

    await client.query("COMMIT");

    console.log(`✅ Saved ${locationData.length} locations and ${eventData.length} events.`);
  } catch (error) {
    await client.query("ROLLBACK");
    throw error;
  } finally {
    client.release();
  }
};

try {
  await resetDatabase();
} catch (error) {
  console.error("⚠️ Reset failed:", error.message);
  process.exitCode = 1;
} finally {
  await pool.end();
}

Get events with their location using JOIN

import { pool } from "../config/database.js";

const getEvents = async (req, res) => {
  try {
    const results = await pool.query(
      `SELECT e.id, e.location_id, e.title, e.description, e.starts_at,
              l.name AS location_name, l.image AS location_image
       FROM events AS e
       JOIN locations AS l ON l.id = e.location_id
       ORDER BY e.starts_at, e.id`,
    );

    res.status(200).json(results.rows);
  } catch (error) {
    console.error("Error loading events:", error.message);
    res.status(500).json({ error: "Unable to load events." });
  }
};

const getEventsByLocation = async (req, res) => {
  const locationId = Number(req.params.locationId);

  if (!Number.isInteger(locationId) || locationId < 1 || locationId > 2147483647) {
    return res.status(400).json({ error: "Invalid location ID." });
  }

  try {
    const locationResults = await pool.query(
      "SELECT id FROM locations WHERE id = $1",
      [locationId],
    );

    if (locationResults.rows.length === 0) {
      return res.status(404).json({ error: "Location not found." });
    }

    const results = await pool.query(
      `SELECT id, location_id, title, description, starts_at
       FROM events
       WHERE location_id = $1
       ORDER BY starts_at, id`,
      [locationId],
    );

    res.status(200).json(results.rows);
  } catch (error) {
    console.error("Error loading events:", error.message);
    res.status(500).json({ error: "Unable to load events for this location." });
  }
};

export default { getEvents, getEventsByLocation };

Many to many with a join table

One student can take many classes, and one class has many students. The middle table stores one row for each pairing.

CREATE TABLE students (
  id SERIAL PRIMARY KEY,
  name VARCHAR(255) NOT NULL
);

CREATE TABLE classes (
  id SERIAL PRIMARY KEY,
  title VARCHAR(255) NOT NULL
);

CREATE TABLE enrollments (
  student_id INTEGER NOT NULL REFERENCES students(id) ON DELETE CASCADE,
  class_id INTEGER NOT NULL REFERENCES classes(id) ON DELETE CASCADE,
  PRIMARY KEY (student_id, class_id)
);

SELECT s.name, c.title
FROM enrollments AS e
JOIN students AS s ON s.id = e.student_id
JOIN classes AS c ON c.id = e.class_id
WHERE c.id = 1;

Column types and rules

CREATE TABLE examples (
  id SERIAL PRIMARY KEY,                       -- auto-increasing id
  name VARCHAR(255) NOT NULL,                  -- short text, required
  email VARCHAR(254) NOT NULL UNIQUE,          -- no duplicates allowed
  bio TEXT,                                    -- long text, optional

  price_cents INTEGER NOT NULL DEFAULT 0,      -- store money as whole cents

SELECT COUNT(*) FROM items;

Adding, changing, and deleting rows

Always include WHERE on UPDATE and DELETE. Without it, every row in the table changes.

INSERT INTO items (title, description) VALUES ('New item', 'Hello');
UPDATE items SET title = 'Changed' WHERE id = 1;
DELETE FROM items WHERE id = 1;

Group and count

SELECT location_id, COUNT(*) AS total
FROM events
GROUP BY location_id
HAVING COUNT(*) > 1
ORDER BY total DESC;

Change a table that has data in it

ALTER TABLE items ADD COLUMN price_cents INTEGER NOT NULL DEFAULT 0;
ALTER TABLE items RENAME COLUMN price_cents TO price;
ALTER TABLE items DROP COLUMN price;
CREATE INDEX items_title_index ON items (title);

Search with user input the safe way

Put the % signs in the value, not in the SQL string.

const search = String(req.query.search ?? "");

const results = await pool.query(
  "SELECT * FROM items WHERE title ILIKE $1 ORDER BY id LIMIT 50",
  [`%${search}%`],
);

Never build SQL by joining strings

The top line lets anyone run their own SQL through your form. The bottom line is safe.

// DANGER: SQL injection
pool.query(`SELECT * FROM users WHERE email = '${req.body.email}'`);

// SAFE: the value travels separately from the SQL
pool.query("SELECT * FROM users WHERE email = $1", [req.body.email]);

Column names turn lowercase

Postgres lowers any name without quotes, so pricePoint becomes pricepoint. That caused my unearthed mismatch. Use snake_case names like price_point, and rename them for React with AS.

const results = await pool.query(
  `SELECT id, price_point AS "pricePoint" FROM gifts`,
);
