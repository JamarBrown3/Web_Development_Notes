How The Rev Login Works

Rev signs people in with a server session, not a token saved in the browser. The password never gets stored, and the browser only holds a cookie that JavaScript cannot read. These files work together, so copy them in the order shown.

  • The browser asks GET /api/auth/csrf and gets a session cookie plus a CSRF token.
  • The login form sends the email, password, and token to POST /api/auth/login.
  • The server checks the rate limits, the token, and the input format.
  • It finds the user and checks the password against the saved Argon2id hash.
  • An admin must also send a 6-digit code from an authenticator app.
  • The server replaces the session id, saves who signed in, and sends back a new CSRF token.
  • On every later request, requireAuth loads the account again and checks that it is still active.
  • When React loads, it asks GET /api/auth/me who is signed in. React only decides what to show. The server is what actually protects the data.

server/security/databaseAccess.js

Stops the server from starting with a database user that has more power than it needs. If someone breaks into the app, they cannot touch your other databases. This check needs an Atlas user with only readWrite on your database.

// Check database infrastructure permissions,
// not the website's user/admin roles.
export function assertApplicationRoles(authInfo, dbName) {
  const roles = authInfo?.authenticatedUserRoles;

  // allowed is true if the connection has exactly one role, readWrite, on the configured database.
  const allowed =
    Array.isArray(roles) &&
    roles.length === 1 &&
    roles[0].role === "readWrite" &&
    roles[0].db === dbName;

  // If the connection has any other roles, throw an error to prevent excessive privileges.
  if (!allowed) {
    const error = new Error(
      "Application database privileges are not restricted.",
    );

    error.code = "EXCESS_DATABASE_PRIVILEGES";
    throw error;
  }
}

server/config/database.js

import mongoose from "mongoose";
import { getDatabaseConfig } from "./env.js";
import { assertApplicationRoles } from "../security/databaseAccess.js";

// Establish a connection and check its database permissions.
export async function connectDatabase() {
  const { uri, dbName } = getDatabaseConfig();

  // Wait for MongoDB before continuing application startup.
  await mongoose.connect(uri, {
    dbName,
    serverSelectionTimeoutMS: 10000, // 10 seconds
    connectTimeoutMS: 10000, // 10 seconds
    socketTimeoutMS: 10000, // 10 seconds
    maxPoolSize: 10, // Limit the number of concurrent connections
  });

  // Ask MongoDB which roles this connection authenticated with.
  const status = await mongoose.connection.db
    .admin()
    .command({ connectionStatus: 1 });

  // Refuse credentials that do not match the app's intended policy.
  assertApplicationRoles(status.authInfo, dbName);
}

// Check that MongoDB can respond to a small command.
export async function pingDatabase() {
  if (mongoose.connection.readyState !== 1) {
    throw new Error("Database is unavailable.");
  }

  // Ping the database to ensure it's responsive.
  const result = await mongoose.connection.db.command(
    { ping: 1 },
    { maxTimeMS: 3000 },
  );

  // If result is not ok then the database is not responding correctly.
  if (result.ok !== 1) {
    throw new Error("Database ping failed.");
  }
  return true;
}

// Close Mongoose connections when the process shuts down.
export async function disconnectDatabase() {
  await mongoose.disconnect();
}

// Translate database errors into messages that do not expose
// connection strings, credentials, or raw driver errors to the user.
export function explainDatabaseError(error) {
  if (error?.code === "EXCESS_DATABASE_PRIVILEGES") {
    return `Use a database account with only readWrite on the configured applications
    database; broad administrator credentials are refused.`;
  }

  // MongoDB authentication errors can be caused by incorrect credentials, expired passwords, or misconfigured database settings.
  if (error?.code === 18 || error?.codeName === "AuthenticationFailed") {
    return "MongoDB authentication failed. Check the database user's credentials locally.";
  }

  // MongoDB connection errors can be caused by network issues, incorrect credentials, or misconfigured database settings.
  if (error?.code === 13 || error?.codeName === "Unauthorized") {
    return "MongoDB denied this operation. Check the database user's permissions.";
  }

  // MongoDB connection errors can be caused by network issues, incorrect credentials, or misconfigured database settings.
  if (error?.name === "MongooseServerSelectionError") {
    return `MongoDB is unreachable. Check Atlas status, your IP access list, DNS, and connection
    settings.`;
  }

  // If the error is not recognized, return a generic message.
  if (!process.env.MONGODB_URI?.trim()) {
    return `MONGODB_URI is missing. Enter it privately in server/.env.`;
  }
  return "MongoDB connection failed. Check the database server, credentials, and connection settings.";
}

server/models/User.js

const userSchema = new mongoose.Schema(
  {
    // username of the user

      type: String,
      required: true,
      trim: true,
      minlength: 3,
      maxlength: 35,
      // letters, digits, "_", "-", "." only (no look-alike Unicode).
      match: USERNAME_PATTERN,
      // uniqueness now ignores capital letters. It is enforced
      // by the "username_case_insensitive_unique" index declared below,
      // so "Sam" and "sam" can no longer both exist.
    },

    email: {
      type: String,
      required: true,

      select: false,

      type: String,
      required: true,

server/models/PasswordResetToken.js

Only a SHA-256 hash of the reset token gets saved, so a copy of the database cannot be used to reset a password.

import mongoose from "mongoose";
// database-wide safety settings; must load before the schema below.
import "../config/mongooseSecurity.js";

const passwordResetTokenSchema = new mongoose.Schema(
  {
    // The account this password-reset request belongs to.
    // A reference does not automatically verify that the user exists.
    userId: {
      type: mongoose.Schema.Types.ObjectId,
      ref: "User",
      required: true,
      index: true,
    },
    // Snapshot from the account when this request was issued. Old requests
    // cannot reset an account after another password change advances its version.
    sessionVersion: {
      type: Number,
      required: true,
      min: 0,
      validate: { validator: Number.isSafeInteger, message: "Invalid session version." },
    },

    // Store ONLY the SHA-256 hash of a cryptographically random token.
    //
    // IMPORTANT: This schema does NOT generate or hash tokens for us.
    // The future controller must do that before creating this document.
    //
    // Never store the original reset token or a password here.

    // Uniqueness prevents duplicate hashes, not multiple requests per account.
    // The recovery service invalidates other requests after a successful reset.
    tokenHash: {
      type: String,
      required: true,
      unique: true,
      select: false,
      match: /^[a-f0-9]{64}$/, // SHA-256 hash format validation
    },

    // The expiration date of the token. After this date, the token is no longer valid.
    expiresAt: {
      type: Date,
      required: true,
    },

    // The date and time when the token was used. If null, the token has not been used yet.
    usedAt: {
      type: Date,
      default: null,
    },
  },
  {
    timestamps: {
      createdAt: true,
      updatedAt: false,
    },

    collection: "passwordResetTokens",
    toJSON: {
      transform(document, result) {
        delete result.tokenHash;
        delete result.__v;
        return result;
      },
    },
  },
);
// TTL means "time to live."
// expireAfterSeconds: 0 tells MongoDB to use the expiresAt date.
//
// Cleanup happens in the background, NOT immediately at expiration.
// The controller MUST still reject tokens whose expiresAt has passed.

// Cleanup can take longer than 60 seconds. Never use deletion timing as the
// security check: the recovery service must check expiration atomically.
passwordResetTokenSchema.index({ expiresAt: 1 }, { expireAfterSeconds: 0 });

const PasswordResetToken = mongoose.model(
  "PasswordResetToken",
  passwordResetTokenSchema,
);

export default PasswordResetToken;

Step 3 Check All Input

server/security/input.js

Treat everything from the browser as untrusted. Each parse function allows a list of fields and rejects anything else, so a request cannot add role: "admin". publicUser is the only account data that goes back to the browser.

// shared username and email rules (also used by models/User.js).
import { EMAIL_PATTERN, USERNAME_PATTERN } from "../models/validators.js";

// Carry an intentional HTTP status to app.js's error handler, which will be used to set the response status code.
export class HttpError extends Error {
  constructor(status, message) {
    super(message);
    this.status = status;
  }
}

/* Inspect nested input without treating it as executable instructions
 or allowing it to become an arbitrary MongoDB query. */

export function assertSafeObject(value, depth = 0) {
  // Limit recursive inspection depth; body-size and traffic limits are separate.
  if (depth > 10) {
    throw new HttpError(400, "Request body is too deeply nested.");
  }
  // Primitives have no nested keys to inspect here.
  // Field-specific helpers must still validate their types and values.
  if (value === null || typeof value !== "object") {
    return;
  }

  for (const key of Object.keys(value)) {
    if (
      key.startsWith("$") ||
      key.includes(".") ||
      ["__proto__", "constructor", "prototype"].includes(key)
    ) {
      throw new HttpError(400, "Request body contains an unsupported field.");
    }

    assertSafeObject(value[key], depth + 1);
  }
}

// Express middleware: inspect parsed JSON before routes use it.
export function rejectUnsafeJson(req, res, next) {
  // No parsed body means this middleware has no object to inspect.
  try {
    if (req.body !== undefined) {
      assertSafeObject(req.body);
    }
    next();
    // Validation errors below are forwarded to the application's error handler.
  } catch (error) {
    next(error);
  }
}

// the URL query string (?page=2&sort=new) is
// untrusted input too. Express 5's "simple" parser turns it into strings or
// arrays of strings, but a future route could still be fooled, for example by
// ?email=a&email=b (an array) or by keys that look like Mongo operators.
// This global check rejects operator-like keys, too many parameters, and
// oversized values before any route sees them. Routes must STILL validate
// the specific parameters they use (type, allowed values, number ranges).
const MAX_QUERY_KEYS = 30;
const MAX_QUERY_VALUES_PER_KEY = 20;
const MAX_QUERY_VALUE_LENGTH = 2000;

export function rejectUnsafeQuery(req, res, next) {
  const query = req.query ?? {};
  const keys = Object.keys(query);
  const unsupported = () =>
    next(new HttpError(400, "Request contains an unsupported query parameter."));

  if (keys.length > MAX_QUERY_KEYS) return unsupported();

  for (const key of keys) {
    if (
      key.length > 100 ||
      key.startsWith("$") ||
      /[.[\]]/.test(key) ||
      ["__proto__", "constructor", "prototype"].includes(key)
    ) {
      return unsupported();
    }
    const values = Array.isArray(query[key]) ? query[key] : [query[key]];
    if (
      values.length > MAX_QUERY_VALUES_PER_KEY ||
      values.some(
        (value) => typeof value !== "string" || value.length > MAX_QUERY_VALUE_LENGTH,
      )
    ) {
      return unsupported();
    }
  }
  next();
}

// Accept only a plain object containing allowed fields.
function fields(body, allowed) {
  // If the request body is not a plain object, throw an error to indicate that the request is invalid.
  if (
    !body ||
    Array.isArray(body) ||
    Object.getPrototypeOf(body) !== Object.prototype
  ) {
    throw new HttpError(400, "Send a JSON object.");
  }

  // Reject unsafe key patterns; the allowlist below rejects unexpected fields.
  assertSafeObject(body);

  // If the request body contains any keys that are not in the allowed list, throw an error to indicate that the request is invalid.
  if (Object.keys(body).some((key) => !allowed.includes(key))) {
    throw new HttpError(400, "Request body contains unsupported fields.");
  }
}

// Reusable text validation.
function string(value, label, min, max, trim = false) {
  // If the value is not a string, throw an error to indicate that the input is invalid.
  if (typeof value !== "string") {
    throw new HttpError(400, `${label} must be text.`);
  }

  /*  result trim or return the value as is, depending on the trim parameter.
  //  If trim is true, the value will be trimmed of whitespace from both ends;
    otherwise, it will be returned as is.
  */
  const result = trim ? value.trim() : value;

  if (result.length < min || result.length > max) {
    throw new HttpError(
      400,
      `${label} must be between ${min} and ${max} characters.`,
    );
  }
  return result;
}
// readEmail function: validate and normalize an email address.
export function readEmail(value) {
  const email = string(value, "Email", 3, 254, true).toLowerCase();

  // Basic format validation does not prove email ownership.
  // EMAIL_PATTERN also refuses control and invisible characters.
  if (!EMAIL_PATTERN.test(email)) {
    throw new HttpError(
      400,
      "Email format is invalid. Enter a valid email address.",
    );
  }
  return email;
}

// Validate a new password's type and length without changing its contents.
export function readNewPassword(value) {
  // Do not trim or silently truncate passwords.
  return string(value, "Password", 15, 128);
}

// names nobody can register, so no visitor can
// pose as site staff. Comparison ignores capital letters and the separators
// "_", ".", and "-", so "Ad_Min" counts as "admin". Add your own names (for
// example your real username) in the RESERVED_USERNAMES environment setting:
//   RESERVED_USERNAMES=yourname,othername
// Reserved names only block NEW registrations; existing accounts keep theirs.
const RESERVED_USERNAMES = [
  "admin", "administrator", "root", "system", "support", "moderator", "mod",
  "staff", "owner", "official", "security", "help", "api", "null",
  "undefined",
];

export function isReservedUsername(
  username,
  extra = process.env.RESERVED_USERNAMES,
) {
  const squash = (name) => name.trim().toLowerCase().replace(/[_.-]/g, "");
  const reserved = new Set(
    [...RESERVED_USERNAMES, ...String(extra ?? "").split(",")]
      .map(squash)
      .filter(Boolean),
  );
  return reserved.has(squash(username));
}

// Validate registration fields and return only accepted values.
export function parseRegistration(body) {
  fields(body, ["username", "email", "password"]);

  // Username validation: must be between 3 and 35 characters, and cannot contain control characters.
  const username = string(body.username, "Username", 3, 35, true);

  //  If the username contains any control characters (ASCII codes 0-31 and 127), throw an error to indicate that the username is invalid.
  if (/[\u0000-\u001f\u007f]/.test(username)) {
    throw new HttpError(400, "Username contains an unsupported character.");
  }

  // letters, digits, "_", "-", and "." only, starting and ending
  // with a letter or digit. This blocks look-alike Unicode names.
  if (!USERNAME_PATTERN.test(username)) {
    throw new HttpError(
      400,
      "Usernames may use letters, numbers, _, -, and . and must start and end with a letter or number.",
    );
  }

  // staff-like names are not available to the public.
  if (isReservedUsername(username)) {
    throw new HttpError(409, "That username is not available.");
  }
  // Return the validated username, normalized email, and unchanged password.
  return {
    username,
    email: readEmail(body.email),
    password: readNewPassword(body.password),
  };
}

// the optional 6-digit two-step code that the
// owner's admin account must send when signing in. Spaces are removed so
// "123 456" (how many authenticator apps display codes) also works.
export function readTotpCode(value) {
  if (value === undefined) return undefined;
  const code = typeof value === "string" ? value.replace(/\s/g, "") : value;
  if (typeof code !== "string" || !/^[0-9]{6}$/.test(code)) {
    throw new HttpError(400, "Enter the 6-digit code from your authenticator app.");
  }
  return code;
}

// Validate login fields; normalize email but preserve the password.
export function parseLogin(body) {
  // "code" is optional and only matters for the admin account.
  fields(body, ["email", "password", "code"]);

  return {
    email: readEmail(body.email),

    /* Check an existing credential without apply the new-password minimym to every login attempt.
     */
    password: string(body.password, "Password", 1, 128),

    code: readTotpCode(body.code),
  };
}

// Validate the email supplied when requesting password-recovery instructions.
export function parseRecovery(body) {
  fields(body, ["email"]);

  return {
    email: readEmail(body.email),
  };
}

// Check token format only; the service checks expiry and single use.
export function readResetToken(value) {
  if (typeof value !== "string" || !/^[a-f0-9]{64}$/.test(value)) {
    throw new HttpError(400, "Reset link is invalid or expired.");
  }
  return value;
}

// Validate the reset token and new password; this request does not take an email.
export function parseReset(body) {
  fields(body, ["token", "newPassword"]);

  return {
    token: readResetToken(body.token),
    newPassword: readNewPassword(body.newPassword),
  };
}

// Return only intentionally public account information.
export function publicUser(user) {
  // Exclude password hashes and internal session-version information.
  return {
    id: user._id.toString(),
    username: user.username,
    role: user.role,
  };
}

Step 4 Password Hashing

server/security/passwords.js

Argon2id is a slow hash made for passwords, and it adds a random salt for you. An unknown email still does the same work against a dummy hash, so a bot cannot find real accounts by timing the response. This is the Rev file with the old scrypt support removed, since a new project has no old passwords to keep working.

import * as argon2 from "argon2";
import { randomBytes } from "node:crypto";
import { HttpError, readNewPassword } from "./input.js";

// memoryCost is measured in KiB: 19456 KiB = 19 MiB.
// OWASP's minimum recommended Argon2id setting.
const ARGON_OPTIONS = {
  type: argon2.argon2id,
  memoryCost: 19456,
  timeCost: 2,
  parallelism: 1,
  hashLength: 32,
};

// Every Argon2id cost setting this app has ever used. Add the old values here
// if you raise the costs, so existing users are not locked out. The first
// entry must always match ARGON_OPTIONS above.
const SUPPORTED_ARGON_PARAMETERS = [{ m: 19456, t: 2, p: 1 }];

function readArgonParameters(encoded) {
  if (typeof encoded !== "string" || encoded.length > 200) {
    return null;
  }

  // The three cost parameters may appear in different orders.
  const match =
    /^\$argon2id\$v=19\$((?:[mtp]=[0-9]{1,6},){2}[mtp]=[0-9]{1,6})\$[A-Za-z0-9+/]{22}\$[A-Za-z0-9+/]{43}$/.exec(
      encoded,
    );

  if (!match) {
    return null;
  }

  // Check that the three parameters are exactly m, t, and p, with no duplicates.
  const parameters = Object.fromEntries(
    match[1].split(",").map((pair) => {
      const [name, value] = pair.split("=");
      return [name, Number(value)];
    }),
  );
  return Object.keys(parameters).length === 3 ? parameters : null;
}

// Do not let a malformed stored hash request unlimited memory or work.
function isSupportedArgonHash(encoded) {
  const parameters = readArgonParameters(encoded);
  return Boolean(
    parameters &&
      SUPPORTED_ARGON_PARAMETERS.some(
        ({ m, t, p }) => parameters.m === m && parameters.t === t && parameters.p === p,
      ),
  );
}

// Wait for a free hashing slot (or fail if the line is too long or too slow).
function acquireSlot() {
  if (active < MAX_CONCURRENT_HASHES) {
    active += 1;
    return Promise.resolve();
  }
  if (waiting.length >= MAX_WAITING_HASHES) {
    return Promise.reject(busy());
  }
  return new Promise((resolve, reject) => {
    const entry = { resolve, timer: null };
    entry.timer = setTimeout(() => {
      const index = waiting.indexOf(entry);
      if (index !== -1) waiting.splice(index, 1);
      reject(busy());
    }, HASH_WAIT_TIMEOUT_MS);
    entry.timer.unref?.();
    waiting.push(entry);
  });
}

// Hand the slot straight to the next waiting request, or free it.
function releaseSlot() {
  const next = waiting.shift();
  if (next) {
    clearTimeout(next.timer);
    next.resolve(); // the slot passes to the next request; "active" is unchanged
  } else {
    active -= 1;
  }
}

async function withPasswordSlot(work) {
  await acquireSlot();

  try {
    return await work();
  } finally {
    // Release the slot even if hashing throws an error.
    releaseSlot();
  }
}

// An unknown account still performs password-verification work.
function getDummyHash() {
  if (!dummyHashPromise) {
    dummyHashPromise = argon2
      .hash(randomBytes(32).toString("hex"), ARGON_OPTIONS)
      .catch((error) => {
        // Allow a later attempt to retry after a library failure.
        dummyHashPromise = undefined;
        throw error;
      });
  }
  return dummyHashPromise;
}

// Used when registering or choosing a NEW password.
export async function hashNewPassword(password) {
  readNewPassword(password);

  // Argon2 generates the random salt for us.
  return withPasswordSlot(() => argon2.hash(password, ARGON_OPTIONS));
}

// Used when checking an EXISTING password.
export async function verifyPassword(password, encoded) {
  if (typeof password !== "string" || password.length < 1 || password.length > 128) {
    return false;
  }

  // A missing hash means the email has no account. It still falls through to
  // the dummy check, so known and unknown emails take the same amount of time.
  return withPasswordSlot(async () => {
    if (isSupportedArgonHash(encoded)) {
      return argon2.verify(encoded, password);
    }

    // Do the verification work, but NEVER accept the password.
    const dummyHash = await getDummyHash();
    await argon2.verify(dummyHash, password);

    return false;
  });
}

// True when a stored hash uses older settings. After a successful sign-in the
// login service re-hashes the password, so old records upgrade themselves.
export function needsRehash(encoded) {
  const parameters = readArgonParameters(encoded);
  if (!parameters) return true;
  return !(
    parameters.m === ARGON_OPTIONS.memoryCost &&
    parameters.t === ARGON_OPTIONS.timeCost &&
    parameters.p === ARGON_OPTIONS.parallelism
  );
}

Step 5 Two Step Codes And Login Limits

server/security/totp.js

The 6-digit codes from Google Authenticator and similar apps, using only Node's built-in crypto. The secret is stored encrypted, so a copy of the database cannot create codes without ADMIN_MFA_KEY.

// two-step sign-in codes for the owner's
// administrator account.
//
// This implements standard time-based one-time passwords (TOTP, RFC 6238),
// the same 6-digit codes shown by Google Authenticator, Microsoft
// Authenticator, Authy, 1Password, and similar apps. Even if someone steals
// or guesses the admin password, they still cannot open admin routes without
// the code from your phone.
//
// Only Node's built-in crypto module is used, so no extra package is needed.

import {
  createCipheriv,
  createDecipheriv,
  createHmac,
  randomBytes,
  timingSafeEqual,
} from "node:crypto";

export const TOTP_STEP_SECONDS = 30;
export const TOTP_DIGITS = 6;

const BASE32_ALPHABET = "ABCDEFGHIJKLMNOPQRSTUVWXYZ234567";

// A new random 160-bit secret, the size recommended for SHA-1 TOTP.
export function generateTotpSecret() {
  return randomBytes(20);
}

// Authenticator apps expect the secret written in Base32 (RFC 4648).
export function base32Encode(buffer) {
  let bits = 0;
  let value = 0;
  let output = "";
  for (const byte of buffer) {
    value = (value << 8) | byte;
    bits += 8;
    while (bits >= 5) {
      output += BASE32_ALPHABET[(value >>> (bits - 5)) & 31];
      bits -= 5;
    }
  }
  if (bits > 0) output += BASE32_ALPHABET[(value << (5 - bits)) & 31];
  return output;
}

// HOTP (RFC 4226): the 6-digit code for one specific counter value.
export function hotp(secret, counter, digits = TOTP_DIGITS) {
  const message = Buffer.alloc(8);
  message.writeBigUInt64BE(BigInt(counter));
  const hmac = createHmac("sha1", secret).update(message).digest();
  const offset = hmac[hmac.length - 1] & 0x0f;
  const binary =
    ((hmac[offset] & 0x7f) << 24) |
    (hmac[offset + 1] << 16) |
    (hmac[offset + 2] << 8) |
    hmac[offset + 3];
  return String(binary % 10 ** digits).padStart(digits, "0");
}

// Which 30-second time window ("step") a moment belongs to.
export function totpStep(now = Date.now()) {
  return Math.floor(now / 1000 / TOTP_STEP_SECONDS);
}

// Check a submitted code. It is accepted in the current 30-second window or
// one window either side (phone clocks drift). lastUsedStep blocks REPLAY:
// a code that was already used, or any older code, is refused.
// Returns the matching step number, or null when the code is not valid.
export function verifyTotp(
  secret,
  code,
  { now = Date.now(), window = 1, lastUsedStep = -1 } = {},
) {
  if (typeof code !== "string" || !/^[0-9]{6}$/.test(code)) return null;
  const current = totpStep(now);
  const submitted = Buffer.from(code);
  let matched = null;
  // Check every candidate window without stopping early, so the response
  // time does not reveal which window matched.
  for (let step = current - window; step <= current + window; step += 1) {
    const expected = Buffer.from(hotp(secret, step));
    if (timingSafeEqual(expected, submitted) && step > lastUsedStep) {
      matched = step;
    }
  }
  return matched;
}

// The link an authenticator app reads (it can be pasted, or turned into a QR code).
export function otpauthUri({ secret, account, issuer = "MyApp" }) {
  const label = `${encodeURIComponent(issuer)}:${encodeURIComponent(account)}`;
  const params = new URLSearchParams({
    secret: base32Encode(secret),
    issuer,
    algorithm: "SHA1",
    digits: String(TOTP_DIGITS),
    period: String(TOTP_STEP_SECONDS),
  });
  return `otpauth://totp/${label}?${params}`;
}

// --- Encryption of the stored secret -------------------------------------
// The TOTP secret is stored ENCRYPTED (AES-256-GCM) in MongoDB. Someone who
// obtains a copy of the database still cannot generate your codes without
// ADMIN_MFA_KEY, which lives only in the server environment.
// The account ID is bound into the encryption, so a secret copied onto a
// different account does not decrypt.

export function readMfaKey(value = process.env.ADMIN_MFA_KEY) {
  const text = value?.trim();
  if (!text) return null;
  if (!/^[a-fA-F0-9]{64}$/.test(text)) {
    throw new Error(
      "ADMIN_MFA_KEY must be 64 hexadecimal characters (32 random bytes).",
    );
  }
  return Buffer.from(text, "hex");
}

export function encryptTotpSecret(secret, key, accountId) {
  const iv = randomBytes(12);
  const cipher = createCipheriv("aes-256-gcm", key, iv);
  cipher.setAAD(Buffer.from(String(accountId)));
  const ciphertext = Buffer.concat([cipher.update(secret), cipher.final()]);
  const tag = cipher.getAuthTag();
  return ["v1", iv, tag, ciphertext]
    .map((part) => (typeof part === "string" ? part : part.toString("base64url")))
    .join(".");
}

export function decryptTotpSecret(stored, key, accountId) {
  const [version, iv, tag, ciphertext] = String(stored).split(".");
  if (version !== "v1" || !iv || !tag || !ciphertext) {
    throw new Error("Unsupported two-step secret format.");
  }
  const decipher = createDecipheriv(
    "aes-256-gcm",
    key,
    Buffer.from(iv, "base64url"),
  );
  decipher.setAAD(Buffer.from(String(accountId)));
  decipher.setAuthTag(Buffer.from(tag, "base64url"));
  return Buffer.concat([
    decipher.update(Buffer.from(ciphertext, "base64url")),
    decipher.final(),
  ]);
}

server/security/accountLoginLimit.js

One login budget for each email, saved in MongoDB. The IP limit alone can be dodged by using many addresses, and it resets on a restart.

// one login budget per normalized email, across IPs and workers.
// The MongoDB counter is separate from sessions and password-reset tokens.
import { createHmac } from "node:crypto";
import { HttpError } from "./input.js";

export function createAccountLoginLimiter({
  collection,
  secret,
  limit = 20,
  windowMs = 15 * 60 * 1000,
  now = () => Date.now(),
} = {}) {
  if (!Number.isSafeInteger(limit) || limit < 1 || limit > 10000 ||
      !Number.isSafeInteger(windowMs) || windowMs < 1000 ||
      typeof secret !== "string" || Buffer.byteLength(secret) < 32) {
    throw new Error("Invalid per-account login-limit configuration.");
  }

  let indexReady;
  function ready() {
    // Missing/unavailable shared storage must never silently disable the limit.
    if (!collection) {
      return Promise.reject(new HttpError(503, "Login protection is temporarily unavailable."));
    }
    indexReady ||= Promise.resolve().then(() => collection.createIndex(
      { expiresAt: 1 },
      { expireAfterSeconds: 0, name: "login_limit_expiry", maxTimeMS: 5000 },
    )).catch(() => {
      indexReady = undefined;
      throw new HttpError(503, "Login protection is temporarily unavailable.");
    });
    return indexReady;
  }

  async function consume(email) {
    await ready();
    const time = now();
    const start = Math.floor(time / windowMs) * windowMs;
    const resetAt = start + windowMs;
    // A keyed digest, not a raw email or an easily guessed unkeyed hash.
    // Every server instance must use the same stable SESSION_SECRET.
    const digest = createHmac("sha256", secret).update("app-login:").update(email).digest("hex");
    const id = `${start}:${digest}`;
    let record;

    for (let attempt = 0; attempt < 2; attempt += 1) {
      try {
        // One atomic increment: simultaneous requests receive different counts.
        // Expiration only cleans up; the bucket ID enforces the time window.
        record = await collection.findOneAndUpdate(
          { _id: id },
          { $inc: { count: 1 }, $setOnInsert: { expiresAt: new Date(resetAt) } },
          { upsert: true, returnDocument: "after", includeResultMetadata: false, maxTimeMS: 5000 },
        );
        break;
      } catch (error) {
        // A first-insert race can hit the unique _id index. Retry that case only.
        if (error.code === 11000 && attempt === 0) continue;
        throw new HttpError(503, "Login protection is temporarily unavailable.");
      }
    }
    if (!Number.isSafeInteger(record?.count) || record.count < 1) {
      throw new HttpError(503, "Login protection is temporarily unavailable.");
    }
    return {
      allowed: record.count <= limit,
      retryAfter: Math.max(1, Math.ceil((resetAt - time) / 1000)),
    };
  }

  return { consume, ready };
}

server/services/authService.js

This is where the account work happens. A duplicate email gets the same answer as a new one, and every failed login gets the same Invalid credentials message. Password reset needs an email service passed in as deliverResetEmail. Until then those routes answer 503. Resets use a MongoDB transaction, which works on Atlas.

import { createHash, randomBytes } from "node:crypto";
import User from "../models/User.js";
import PasswordResetToken from "../models/PasswordResetToken.js";

import {
  HttpError,
  parseRegistration,
  parseLogin,
  parseRecovery,
  parseReset,
} from "../security/input.js";

// passwords.js exports hashNewPassword, not
// hashPassword. Importing a name that does not exist is a SyntaxError in ES
// modules, so this line stopped the entire server from starting.
import {
  hashNewPassword,
  needsRehash,
  verifyPassword,
} from "../security/passwords.js";

// marks operators written by THIS server code as safe (see config/mongooseSecurity.js).
import { trusted } from "../config/mongooseSecurity.js";
// case-insensitive username comparison rule.
import { USERNAME_COLLATION } from "../models/validators.js";
// two-step codes for the owner's admin account.
import { decryptTotpSecret, readMfaKey, verifyTotp } from "../security/totp.js";

export const RESET_LIFETIME_MS = 15 * 60 * 1000;

// limits on reset emails, stored in MongoDB.
// The in-memory rate limiters in auth.js reset whenever the server restarts.
// A free Render server sleeps and restarts often, and in testing 3 restarts
// allowed 9 reset emails to the same inbox. These limits are checked against
// the saved reset records, so they survive restarts:
//   - at most 3 reset emails per account per hour, and
//   - at most 1 reset email per account per minute.
// Requests over the limit are silently skipped; the person still sees the
// same generic "If an account exists..." message, so nothing is revealed.
// (Both numbers can be changed per service with the options below.)
export const RESET_EMAILS_PER_HOUR = 3;
export const RESET_EMAIL_MIN_GAP_MS = 60 * 1000;

// Reset tokens are cryptographically random, unlike human passwords.
// Store their digest, not the usable token itself.
export const tokenDigest = (token) =>
  createHash("sha256").update(token).digest("hex");

export function createAuthService({
  UserModel = User,
  TokenModel = PasswordResetToken,
  connection = UserModel.db,
  deliverResetEmail = null,
  resetOrigin = "http://localhost:5173",
  // key that decrypts the admin's two-step secret (ADMIN_MFA_KEY).
  mfaKey = readMfaKey(),
  // database-backed reset-email limits (see the top of this file).
  resetEmailsPerHour = RESET_EMAILS_PER_HOUR,
  resetEmailMinGapMs = RESET_EMAIL_MIN_GAP_MS,
} = {}) {
  // Use a server-configured frontend origin when building reset links.
  // Never build these links from a client-supplied Host header.
  const origin = new URL(resetOrigin);

  if (
    !["https:", "http:"].includes(origin.protocol) ||
    origin.username ||
    origin.password ||
    origin.pathname !== "/" ||
    origin.search ||
    origin.hash
  ) {
    throw new Error(
      "Configure a trusted frontend origin for password recovery.",
    );
  }

  // REGISTER: validate, hash, then save explicitly allowed fields.
  async function register(body) {
    const { username, email, password } = parseRegistration(body);

    // usernames are public (they appear on
    // posts), so it is safe and friendlier to say when one is taken. The
    // check ignores capital letters: "Sam" blocks "sam". It happens BEFORE
    // anything involving the email, so the answer never depends on whether
    // the email is registered. Duplicate EMAILS still get the generic 202.
    const usernameTaken = await UserModel.exists({ username }).collation(
      USERNAME_COLLATION,
    );
    if (usernameTaken) {
      throw new HttpError(409, "That username is not available.");
    }

    const passwordHash = await hashNewPassword(password);

    try {
      await UserModel.create({
        username,
        email,
        passwordHash,
        role: "user",
        status: "active",
        sessionVersion: 0,
      });
    } catch (error) {
      // Duplicate email/username gets the same public response.
      // Other database failures must still reach the error handler.
      if (error.code !== 11000) {
        throw error;
      }
    }
  }

  // LOGIN: validate input and check the stored password hash.
  async function authenticate(body) {
    const { email, password, code } = parseLogin(body);

    // passwordHash is normally excluded by the User schema.
    // This trusted authentication operation explicitly requests it.
    // the encrypted two-step fields are hidden the same way.
    const user = await UserModel.findOne({ email }).select(
      "+passwordHash +mfa.secretEnc +mfa.lastUsedStep",
    );

    const matches = await verifyPassword(password, user?.passwordHash);

    if (
      !matches ||
      !user ||
      user.status !== "active" ||
      !Number.isSafeInteger(user.sessionVersion)
    ) {
      throw new HttpError(401, "Invalid credentials.");
    }

    // an account with two-step sign-in turned on (the owner's
    // admin account, set up with scripts/create-admin.js) must also send the
    // current 6-digit code. A wrong, missing, or reused code gets the SAME
    // "Invalid credentials" answer as a wrong password. Because this check
    // happens before any session exists, every session on that account has
    // passed the code check.
    // an admin without an enrolled second factor cannot sign in.
    // Merely changing the stored role must not create an admin login path.
    const mfaEnabled = Boolean(user.mfa?.enabledAt && user.mfa?.secretEnc);
    if (user.role === "admin" && !mfaEnabled) {
      throw new HttpError(401, "Invalid credentials.");
    }
    let mfaVerified = false;
    if (mfaEnabled) {
      const codeAccepted = await checkTwoStepCode(user, code);
      if (!codeAccepted) {
        throw new HttpError(401, "Invalid credentials.");
      }
      mfaVerified = true;
    }

    // quietly upgrade old password hashes (legacy scrypt or older
    // Argon2 costs) now that we know the password is correct. Best effort: a
    // failure here must never block a valid sign-in.
    if (needsRehash(user.passwordHash)) {
      try {
        const upgraded = await hashNewPassword(password);
        await UserModel.updateOne(
          { _id: user._id, passwordHash: user.passwordHash },
          { $set: { passwordHash: upgraded } },
        );
      } catch {
        // Keep the old (still valid) hash; the next sign-in will try again.
      }
    }

    // return explicit, server-generated authentication assurance.
    return {
      user,
      mfaVerified,
      mfaEnabledAt: mfaVerified ? new Date(user.mfa.enabledAt).getTime() : null,
    };
  }

  // Preserve the existing service helper for callers that only need the user.
  async function login(body) {
    return (await authenticate(body)).user;
  }

  // verify the 6-digit code and claim it so it can't be reused.
  async function checkTwoStepCode(user, code) {
    if (!mfaKey || !code) return false;

    let secret;
    try {
      secret = decryptTotpSecret(user.mfa.secretEnc, mfaKey, user._id.toString());
    } catch {
      return false; // wrong ADMIN_MFA_KEY or a damaged record: fail closed
    }

    const step = verifyTotp(secret, code, {
      lastUsedStep: user.mfa.lastUsedStep ?? -1,
    });
    if (step === null) return false;

    // Record the code's time window in the database in ONE atomic update.
    // If two requests race with the same code, only one can win.
    const claim = await UserModel.updateOne(
      {
        _id: user._id,
        $or: [
          { "mfa.lastUsedStep": trusted({ $lt: step }) },
          { "mfa.lastUsedStep": trusted({ $exists: false }) },
        ],
      },
      { $set: { "mfa.lastUsedStep": step } },
    );
    return claim.matchedCount === 1;
  }

  // REQUEST RESET: create a temporary token and deliver its link.
  async function requestReset(body) {
    const { email } = parseRecovery(body);

    if (!deliverResetEmail) {
      throw new HttpError(503, "Password recovery is not configured yet.");
    }

    // email is hidden from normal queries now, so this trusted
    // recovery operation asks for it explicitly.
    const user = await UserModel.findOne({
      email,
      status: "active",
    }).select("+email");

    // The HTTP route gives generic feedback either way.
    if (!user) {
      return;
    }

    // database-backed limits that survive server restarts.
    const now = Date.now();
    const recent = await TokenModel.find({
      userId: user._id,
      createdAt: trusted({ $gt: new Date(now - 60 * 60 * 1000) }),
    })
      .select("createdAt")
      .lean();
    const tooMany = recent.length >= resetEmailsPerHour;
    const tooSoon = recent.some(
      (record) => now - new Date(record.createdAt).getTime() < resetEmailMinGapMs,
    );
    if (tooMany || tooSoon) {
      return; // silently skip: same public response, no new email
    }

    const rawToken = randomBytes(32).toString("hex");

    const record = await TokenModel.create({
      userId: user._id,
      sessionVersion: user.sessionVersion,
      tokenHash: tokenDigest(rawToken),
      expiresAt: new Date(Date.now() + RESET_LIFETIME_MS),
    });

    // the token now goes in the URL FRAGMENT (#token=...) instead
    // of the query string (?token=...). Browsers never send the part after
    // "#" to any server, so the secret stays out of Netlify/Render access
    // logs and out of Referer headers. The future reset page reads it with:
    //   new URLSearchParams(window.location.hash.slice(1)).get("token")
    const resetUrl = new URL("/reset-password", origin);
    resetUrl.hash = new URLSearchParams({ token: rawToken }).toString();

    try {
      // Only a trusted, server-supplied email adapter receives this.
      // Never return or log this URL in the public API.
      await deliverResetEmail({
        to: user.email,
        resetUrl: resetUrl.toString(),
      });
    } catch {
      // Remove the token record if delivery reports failure.
      await TokenModel.deleteOne({ _id: record._id });

      throw new HttpError(503, "Password recovery is temporarily unavailable.");
    }
  }

  // RESET PASSWORD: consume the token and change the password together.
  async function resetPassword(body) {
    const { token, newPassword } = parseReset(body);

    if (!deliverResetEmail) {
      throw new HttpError(503, "Password recovery is not configured yet.");
    }

    const tokenHash = tokenDigest(token);

    // make sure the link is real BEFORE doing
    // the expensive Argon2 hash. Otherwise a bot could post random tokens
    // and make the server burn CPU on hashes that are thrown away. The
    // transaction below still re-checks everything atomically.
    const looksValid = await TokenModel.exists({
      tokenHash,
      usedAt: null,
      expiresAt: trusted({ $gt: new Date() }),
    });
    if (!looksValid) {
      throw new HttpError(400, "Reset link is invalid or expired.");
    }

    const passwordHash = await hashNewPassword(newPassword);

    const session = await connection.startSession();

    try {
      await session.withTransaction(async () => {
        const now = new Date();

        // Require the exact digest, unused token, and future expiry.
        // Claim the token within this transaction.
        const record = await TokenModel.findOneAndUpdate(
          {
            tokenHash,
            usedAt: null,
            // trusted() marks this server-written operator as safe.
            expiresAt: trusted({ $gt: now }),
          },
          {
            $set: { usedAt: now },
          },
          {
            session,
            returnDocument: "after",
            runValidators: true,
          },
        );

        if (
          !record ||
          !Number.isSafeInteger(record.sessionVersion) ||
          record.sessionVersion < 0 ||
          record.sessionVersion >= Number.MAX_SAFE_INTEGER
        ) {
          throw new HttpError(400, "Reset link is invalid or expired.");
        }

        // Change only the active account with the expected version.
        // Incrementing the version invalidates its existing logins.
        const result = await UserModel.updateOne(
          {
            _id: record.userId,
            status: "active",
            sessionVersion: record.sessionVersion,
          },
          {
            $set: { passwordHash },
            $inc: { sessionVersion: 1 },
          },
          {
            session,
            runValidators: true,
          },
        );

        if (result.matchedCount !== 1) {
          throw new HttpError(400, "Reset link is invalid or expired.");
        }

        // Invalidate all outstanding reset links for this account.
        await TokenModel.deleteMany({ userId: record.userId }, { session });
      });
    } finally {
      await session.endSession();
    }
  }

  return {
    register,
    login,
    authenticate,
    requestReset,
    resetPassword,
    recoveryEnabled: Boolean(deliverResetEmail),
    // lets startup warn when admin two-step codes can't be checked.
    mfaConfigured: Boolean(mfaKey),
    UserModel,
  };
}

    secret = randomBytes(32).toString("hex");
    console.warn(
      "Warning: Using a temporary session secret. Restarting the server will invalidate all sessions.",
    );
  }

  // Reject a missing, non-text, or too-short signing key.
  // Length alone does not prove randomness; generate production secrets securely.
  if (typeof secret !== "string" || Buffer.byteLength(secret) < 32) {
    throw new Error(
      "Set a strong SESSION_SECRET before deploying authentication.",
    );
  }

  // Validate the SameSite configuration for session cookies.
  if (
    !["strict", "lax", "none"].includes(sameSite) ||
    (sameSite === "none" && !production)
  ) {
    throw new Error("Invalid sessions SameSite configuration.");
  }

  // Normalize the configured frontend URL for Origin comparisons.
  const allowedOrigin = new URL(clientOrigin).origin;

  // Production authentication requires and must use an HTTPS frontend origin.
  if (production && !allowedOrigin.startsWith("https://")) {
    throw new Error(
      "Production authentication requires a HTTPS frontend origin.",
    );
  }
  // Require a supplied store or a connection from which to create a store.
  // A supplied test store may be in memory; production uses MongoDB here.
  if (!store && !connection) {
    throw new Error("Authentication requires a persistent session store.");
  }

  // SECTION 2: Configure persistent sessions.

  // Reuse the already-connected MongoDB client.
  store ||= MongoStore.create({
    client: connection.getClient(),
    dbName: connection.name,
    collectionName: "sessions",
    ttl: 1800,
    touchAfter: 60,
  });

  // Set the session cookie name based on the environment.
  // name the cookie with the "__Host-" prefix to enforce secure cookie attributes in production.
  const name = production ? "__Host-app.sid" : "app.sid";

  // Configure the session cookie with security settings.
  const cookie = {
    httpOnly: true,
    secure: production,
    sameSite,
    path: "/",
    maxAge: 30 * 60 * 1000,
  };

  // Create the session middleware with the configured options.
  const sessionMiddleware = session({
    name,
    secret,
    store,
    resave: false,
    saveUninitialized: false,
    rolling: true,
    cookie,
  });

  const router = Router();

  // count known and unknown addresses identically, before hashing.
  accountLoginLimiter ||= createAccountLoginLimiter({
    collection: loginAccountCollection,
    secret,
    limit: accountLoginLimit,
  });

  async function limitAccountLogin(req, res, next) {
    try {
      req.loginInput = parseLogin(req.body);
      const result = await accountLoginLimiter.consume(req.loginInput.email);
      if (!result.allowed) {
        res.set("Retry-After", String(result.retryAfter));
        throw new HttpError(429, "Too many attempts. Please try again later.");
      }
      next();
    } catch (error) {
      next(error);
    }
  }

  // SECTION 3: Limit repeated authentication requests.

  // Separate instances create separate endpoint counters.

  // Limit repeated requests per IP; this reduces abuse but does not stop all bots.
  const makeLimit = () =>
    rateLimit({
      windowMs: 15 * 60 * 1000,
      limit: rateLimitMax,
      standardHeaders: "draft-8",
      legacyHeaders: false,
      message: {
        error: "Too many attempts. Please try again later.",
      },
    });

  // Create separate rate limiters for login, registration, and password reset endpoints to prevent abuse.
  const loginLimit = makeLimit();
  const registerLimit = makeLimit();
  const resetLimit = makeLimit();

  // Create a rate limiter for CSRF token requests to prevent abuse.
  const csrfLimit = rateLimit({
    windowMs: 15 * 60 * 1000,
    limit: 60,
    standardHeaders: "draft-8",
    legacyHeaders: false,
    message: {
      error: "Too many attempts. Please try again later.",
    },
  });

  const recoveryAccountLimit = rateLimit({
    windowMs: 60 * 60 * 1000, // 1 hour
    limit: 3,

    // avoid using raw email addresses as counter keys.
    keyGenerator: (req) => {
      return createHash("sha256").update(req.recoveryInput.email).digest("hex");
    },

    standardHeaders: "draft-8",
    legacyHeaders: false,
    message: {
      error: "Too many attempts. Please try again later.",
    },
  });

  // SECTION 4: KEEP recovery work bounded.

  // This queue is local to one running Node process.
  // A scaled deployment needs a durable worker and shared limits.
  const recoveryQueue = [];
  let recoveryWorking = false;

  // Process queued password-recovery requests one at a time to avoid overwhelming the email service.
  async function drainRecovery() {
    if (recoveryWorking) return;

    recoveryWorking = true;
    // try to process all queued recovery requests, handling any errors that occur during processing.
    try {
      while (recoveryQueue.length) {
        const input = recoveryQueue.shift();
        // Attempt to process the password recovery request using the provided input.
        try {
          await service.requestReset(input);
        } catch {
          console.warn("A queued password-recovery could not be completed.");
        }
      }
      // Always release the worker flag, including after an error.
    } finally {
      recoveryWorking = false;
    }
  }

  // SECTION 5: Apply authentication-response protections.

  // Prevent caching of sensitive authentication responses and enforce a strict referrer policy to enhance security.
  router.use((req, res, next) => {
    res.set("Cache-Control", "no-store");
    res.set("Referrer-Policy", "no-referrer");
    next();
  });

  // Reject a supplied Origin that differs from the configured frontend origin.
  // Requests without Origin are not rejected here; this is not authentication.
  router.use((req, res, next) => {
    const origin = req.get("Origin");

    if (origin && origin !== allowedOrigin) {
      return next(new HttpError(403, "Request origin is not allowed."));
    }
    next();
  });

  // SECTION 6: Validate CSRF tokens.

  // Express middleware: inspect CSRF token before routes use it.
  function csrf(req, res, next) {
    const supplied = req.get("X-CSRF-Token");
    const expected = req.session?.csrfToken;

    // Reject missing, malformed, or mismatched CSRF tokens.
    // A valid token is not proof that the requester is human.
    if (
      typeof supplied !== "string" ||
      typeof expected !== "string" ||
      !/^[a-f0-9]{64}$/.test(supplied) ||
      !/^[a-f0-9]{64}$/.test(expected) ||
      supplied.length !== expected.length ||
      !timingSafeEqual(Buffer.from(supplied), Buffer.from(expected))
    ) {
      return next(new HttpError(403, "Refresh the form and try again."));
    }
    next();
  }

  // future API mutations are protected even if their router forgets
  // to add csrf itself. GET/HEAD/OPTIONS must remain read-only in route code.
  function csrfForMutations(req, res, next) {
    if (["GET", "HEAD", "OPTIONS"].includes(req.method)) return next();
    const origin = req.get("Origin");
    if (origin && origin !== allowedOrigin) {
      return next(new HttpError(403, "Request origin is not allowed."));
    }
    return csrf(req, res, next);
  }

  // SECTION 7: Authenticate the current request.

  // Express middleware: inspect session before routes use it.
  async function requireAuth(req, res, next) {
    try {
      const auth = req.session?.auth;
      if (
        !auth ||
        typeof auth.userId !== "string" ||
        !/^[a-f0-9]{24}$/.test(auth.userId) ||
        !Number.isSafeInteger(auth.sessionVersion) ||
        !Number.isFinite(auth.authenticatedAt) ||
        Date.now() - auth.authenticatedAt > 8 * 60 * 60 * 1000
      ) {
        throw new HttpError(401, "Please sign in.");
      }

      // Re-check the actual account, not a browser-supplied role.
      const user = await service.UserModel.findById(auth.userId);
      // If the user is not found, inactive, or has a different session version than the one stored in the session, destroy the session and clear the cookie to force re-authentication.
      if (
        !user ||
        user.status !== "active" ||
        user.sessionVersion !== auth.sessionVersion
      ) {
        await new Promise((resolve, reject) => {
          req.session.destroy((error) => (error ? reject(error) : resolve()));
        });

        // Remove this matching browser cookie after invalidating the server session.
        // Cookie removal does not erase logs or remove every trace of a session.
        res.clearCookie(name, {
          httpOnly: true,
          secure: production,
          sameSite,
          path: "/",
        });
        throw new HttpError(401, "Please sign in.");
      }
      // If the user is found and valid, attach the user object to the request
      // for use in subsequent middleware or route handlers, and call next() to proceed.
      req.authUser = user;
      next();
    } catch (error) {
      next(error);
    }
  }

  // Authenticate first, then check administrative privileges.
  // Check the administrator role only after the session/account is authenticated.
  async function requireAdmin(req, res, next) {
    return requireAuth(req, res, (error) => {
      if (error) return next(error);

      // request authorized user and sees if they have the "admin" role.
      //  If not, return a 403 error to indicate that administrative access is required.
      // database role alone is insufficient. This exact session
      // must have signed in AS an administrator and passed the MFA check.
      // Bind proof to enrollment so disabling/re-enrolling invalidates it.
      const auth = req.session.auth;
      const enabledAt = new Date(req.authUser.mfa?.enabledAt ?? 0).getTime();
      if (
        req.authUser.role !== "admin" ||
        auth.roleAtLogin !== "admin" ||
        auth.mfaVerified !== true ||
        !Number.isFinite(enabledAt) || enabledAt <= 0 ||
        auth.mfaEnabledAt !== enabledAt
      ) {
        return next(new HttpError(403, "Administrative access is required."));
      }
      next();
    });
  }

  // SECTION 8: Authentication routes.

  // Provide a CSRF token for forms that require it, generating one if it doesn't already exist in the session.
  router.get("/csrf", csrfLimit, (req, res) => {
    req.session.csrfToken ||= randomBytes(32).toString("hex");

    res.json({
      csrfToken: req.session.csrfToken,
    });
  });

  // Return the currently authenticated user's public information, if any, or an error if the user is not authenticated.
  router.get("/me", requireAuth, (req, res) => {
    res.json({
      user: publicUser(req.authUser),
    });
  });

  // Delegate registration to the service and return generic feedback.
  // A 202 response does not prove that a new account was created.
  router.post("/register", registerLimit, csrf, async (req, res) => {
    await service.register(req.body);

    res.status(202).json({
      message: "Registration request received. You can try signing in.",
    });
  });

  // Accept a login request, validate it, and create a new session if the credentials are correct. Rate-limited to prevent brute-force attacks.
  router.post("/login", loginLimit, csrf, limitAccountLogin, async (req, res) => {
    // The service creates assurance only after actually verifying the code.
    // It is never copied from req.body or a role/MFA flag in the database.
    const { user, mfaVerified, mfaEnabledAt } = await service.authenticate(req.loginInput);

    // Replace the pre-login session identifier.
    await new Promise((resolve, reject) => {
      req.session.regenerate((error) => (error ? reject(error) : resolve()));
    });

    req.session.auth = {
      userId: user._id.toString(),
      sessionVersion: user.sessionVersion,
      authenticatedAt: Date.now(),
      roleAtLogin: user.role,
      mfaVerified: mfaVerified === true,
      mfaEnabledAt,
    };

    // The old session's CSRF token is no longer used.
    req.session.csrfToken = randomBytes(32).toString("hex");

    // Finish saving before the next request uses the session.
    await new Promise((resolve, reject) => {
      req.session.save((error) => (error ? reject(error) : resolve()));
    });

    // Return the authenticated user's public information and a new CSRF token for subsequent requests.
    res.json({
      user: publicUser(user),
      csrfToken: req.session.csrfToken,
    });
  });

  // Accept a logout request, destroy the session, and clear the session cookie to log the user out.
  router.post("/logout", csrf, async (req, res) => {
    await new Promise((resolve, reject) => {
      req.session.destroy((error) => (error ? reject(error) : resolve()));
    });

    res.clearCookie(name, {
      httpOnly: true,
      secure: production,
      sameSite,
      path: "/",
    });

    res.json({
      message: "Signed out.",
    });
  });

  // Validate and enqueue recovery input when delivery is configured.
  // The worker must also be scheduled for queued requests to be processed.

  router.post(
    "/forgot-password",
    resetLimit,
    csrf,

    // Validate the body before the account-based limiter uses it.
    (req, res, next) => {
      try {
        req.recoveryInput = parseRecovery(req.body);
        next();
      } catch (error) {
        next(error);
      }
    },
    recoveryAccountLimit,
    (req, res) => {
      if (!service.recoveryEnabled) {
        throw new HttpError(503, "Password recovery is not configured yet.");
      }

      // Keep the process-local queue bounded.
      if (recoveryQueue.length >= 20) {
        throw new HttpError(429, "Too many attempts. Please try again later.");
      }
      recoveryQueue.push(req.recoveryInput);

      // Do not reveal whether this email belongs to an account.
      res.status(202).json({
        message: "If an account exists, a recovery email will be sent.",
      });

      // Start processing the queue after sending the response.
      setImmediate(() => {
        void drainRecovery();
      });
    },
  );

  // Accept a password reset request, validate it, and reset the user's password if the token is valid. Rate-limited to prevent abuse.
  router.post("/reset-password", resetLimit, csrf, async (req, res) => {
    await service.resetPassword(req.body);
    await new Promise((resolve, reject) => {
      req.session.destroy((error) => (error ? reject(error) : resolve()));
    });

    // Remove the matching browser cookie after the server session is destroyed.
    res.clearCookie(name, {
      httpOnly: true,
      secure: production,
      sameSite,
      path: "/",
    });

    // Return a message indicating that the password has been changed and the user needs to sign in again.
    res.json({
      message: "Password changed. Please sign in again.",
    });
  });

  // SECTION 9: Return the router and middleware for use in the main application.
  return {
    router,
    sessionMiddleware,
    requireAuth,
    requireAdmin,
    csrf,
    csrfForMutations,
    ready: () => accountLoginLimiter.ready?.(),
    cookieOptions: {
      name,
      ...cookie,
    },
  };
}

Step 8 Put The Server Together

server/app.js

The order here matters. Security headers and the rate limit run first, so a bad request gets turned away before it costs any work. New routers go in through mountRoutes, so they always sit behind every protection.

import express from "express";
import cors from "cors";
import helmet from "helmet";
import { randomUUID } from "node:crypto";
import { rateLimit } from "express-rate-limit";
import { pingDatabase } from "./config/database.js";
// rejectUnsafeQuery checks the URL query string the way
// rejectUnsafeJson already checks JSON bodies.
import { HttpError, rejectUnsafeJson, rejectUnsafeQuery } from "./security/input.js";

// Separating app creation from app.listen makes HTTP behavior testable
// without needing a live Atlas cluster for every unit test.
export function createApp({
  checkDatabase = pingDatabase,
  clientOrigin = process.env.CLIENT_ORIGIN || "http://localhost:5173",
  auth = null,
  trustProxy = false,
  // requests allowed per visitor IP per 5 minutes, across the whole API.
  globalRateLimitMax = 300,
  // a function that adds your future routers, e.g.
  //   mountRoutes: (app) => { app.use("/api/posts", postRoutes); }
  // It runs AFTER all the security middleware below and BEFORE the 404 handler.
  mountRoutes = null,
} = {}) {
  const app = express();
  app.disable("x-powered-by");
  // Configure proxy trust explicitly in deployment; never trust arbitrary
  // X-Forwarded-* headers merely because the client sends them.
  app.set("trust proxy", trustProxy);
  // lock in Express 5's "simple" query parser. It never builds
  // nested objects, so ?email[$ne]=x stays a harmless literal key. (The
  // "extended" parser would turn it into { email: { $ne: "x" } }.)
  app.set("query parser", "simple");

  // give every request a random ID. It is sent
  // back in the X-Request-Id header and printed with any server error, so you
  // can match a user's "something went wrong" to the right log line. An ID
  // supplied by the client is ignored (it could be used to fake log entries).
  app.use((req, res, next) => {
    req.id = randomUUID();
    res.set("X-Request-Id", req.id);
    next();
  });

  app.use(helmet());

  // compare against the normalized origin (scheme://host:port),
  // matching what security/auth.js does. Before, a CLIENT_ORIGIN written with
  // a trailing slash ("https://site.netlify.app/") silently blocked the real
  // frontend, because browsers never send that slash in the Origin header.
  const allowedOrigin = new URL(clientOrigin).origin;

  // Only the configured frontend gets credentialed CORS response headers.
  // CORS is a browser policy, NOT a replacement for authentication or CSRF checks.
  app.use(cors({
    origin(origin, callback) {
      callback(null, origin === allowedOrigin);
    },
    credentials: true,
  }));

  // one overall speed limit for every visitor IP,
  // in addition to the stricter login/register/reset limits in auth.js. It
  // runs before the body is read or the database is touched, so a flooding
  // bot is turned away cheaply. It comes AFTER cors() so the website can
  // still read the 429 response. /api/health is skipped so the hosting
  // provider's health checks never get the server marked as "down".
  // This limit lives in this server's memory: it resets on restart and is
  // per server instance. Large floods (DDoS) must be stopped before they
  // reach Render, for example by Cloudflare. See the security report.
  app.use(rateLimit({
    windowMs: 5 * 60 * 1000,
    limit: globalRateLimitMax,
    standardHeaders: "draft-8",
    legacyHeaders: false,
    skip: (req) => req.path === "/api/health",
    message: { error: "Too many requests. Please slow down and try again shortly." },
  }));

  // API answers may contain private data, so browsers, proxies,
  // and CDNs must never store them. (Auth routes already did this; now every
  // /api route does.) A future PUBLIC, cacheable route can override it.
  app.use("/api", (req, res, next) => {
    res.set("Cache-Control", "no-store");
    next();
  });

  app.use(express.json({ limit: "100kb" }));
  app.use(rejectUnsafeJson);
  // operator-looking or oversized query parameters are refused.
  app.use(rejectUnsafeQuery);

  if (auth) {
    app.use(auth.sessionMiddleware);
    // protect every unsafe API method BEFORE auth or future routers.
    app.use("/api", auth.csrfForMutations);
    app.use("/api/auth", auth.router);
    // Dashboard/content routes are still a future feature, but their prefixes
    // cannot accidentally become public when added later.
    app.use(["/api/admin", "/api/dashboard"], auth.requireAdmin);
    // areas for ANY signed-in account.
    //   /api/account : a member's own profile, posts, settings
    //   /api/demo    : the visitor "try my dashboard + AI summary + PDF" demo.
    // Put each future router under the right prefix and it is protected
    // automatically.
    app.use(["/api/account", "/api/demo"], auth.requireAuth);
  }

  app.get("/", (req, res) => {
    res.json({ message: "API is running." });
  });

  // Liveness checks Express. Readiness below also checks MongoDB.
  app.get("/api/health", (req, res) => {
    res.set("Cache-Control", "no-store");
    res.json({ status: "ok", service: "api" });
  });

  app.get("/api/health/database", async (req, res) => {
    res.set("Cache-Control", "no-store");
    try {
      await checkDatabase();
      res.json({ status: "ok", database: "connected" });
    } catch {
      // Never expose the URI, password, server address, or driver stack.
      res.status(503).json({ status: "error", database: "unavailable" });
    }
  });

  // future routers plug in here, behind every protection above.
  // Routes are refused without authentication middleware, so a test or a
  // mistake can never run them with the admin/account guards missing.
  if (mountRoutes) {
    if (!auth) throw new Error("mountRoutes requires authentication middleware (auth).");
    mountRoutes(app, { auth });
  }

  app.use((req, res) => {
    res.status(404).json({ error: "Route not found." });
  });

  app.use((error, req, res, next) => {
    if (res.headersSent) return next(error);
    if (error instanceof HttpError) return res.status(error.status).json({ error: error.message });
    if (error.type === "entity.too.large") {
      return res.status(413).json({ error: "Request body is too large." });
    }
    if (error.type === "entity.parse.failed") {
      return res.status(400).json({ error: "Invalid JSON request body." });
    }
    // other problems with the REQUEST itself (unsupported charset
    // or compression, an aborted upload) are the client's mistake: 4xx, not
    // 500. Their messages come from body-parser and are marked safe to show.
    if (error.expose === true && Number.isInteger(error.status) && error.status >= 400 && error.status < 500) {
      return res.status(error.status).json({ error: "The request could not be processed." });
    }
    // database validation/cast errors mean the submitted data was
    // invalid. Answer 400 without echoing field values or schema details.
    if (["ValidationError", "CastError", "StrictModeError"].includes(error.name)) {
      return res.status(400).json({ error: "The request contains invalid data." });
    }
    // log unexpected errors on the server WITHOUT the request body
    // (it may contain passwords) and without the message in production
    // (MongoDB messages can include emails, e.g. duplicate-key errors).
    console.error(JSON.stringify({
      level: "error",
      requestId: req.id,
      method: req.method,
      path: req.path,
      name: error?.name,
      code: error?.code,
      ...(process.env.NODE_ENV === "production" ? {} : { stack: error?.stack }),
    }));
    res.status(500).json({ error: "An unexpected server error occurred.", requestId: req.id });
  });

  return app;
}

Startup checks the settings, connects to the database, builds the unique indexes, and only then opens the port. HTTP_LIMITS cuts off clients that send a request a few bytes at a time.

import http from "node:http";
import mongoose from "mongoose";
import { createApp } from "./app.js";
import { getPort, getTrustProxy } from "./config/env.js";
import {
  connectDatabase,
  pingDatabase,
  disconnectDatabase,
  explainDatabaseError,
} from "./config/database.js";
import { createAuth } from "./security/auth.js";
import { createAuthService } from "./services/authService.js";

let server;
let stopping = false;

// Limits for slow or stalled clients (a "Slowloris" attack).
const HTTP_LIMITS = {
  headersTimeout: 20_000, // the request headers must arrive within 20 s
  requestTimeout: 30_000, // the whole request must arrive within 30 s
  connectionsCheckingInterval: 5_000, // how often Node checks those deadlines
};
// Render's proxy keeps idle connections open for up to 120 s. Node must keep
// them slightly longer, or visitors see a random 502 error.
const PRODUCTION_KEEP_ALIVE_MS = 121_000;

async function shutdown() {
  if (stopping) return;
  stopping = true;
  // Bound shutdown if a client holds an HTTP connection open.
  const timer = setTimeout(() => process.exit(1), 10000);
  timer.unref();
  try {
    if (server) await new Promise((resolve) => server.close(resolve));
    await disconnectDatabase();
  } finally {
    clearTimeout(timer);
  }
}

process.once("SIGINT", () => {
  void shutdown();
});
process.once("SIGTERM", () => {
  void shutdown();
});

// Remember which startup step failed, so the error points at the real cause.
let stage = "configuration";

try {
  const port = getPort();
  const trustProxy = getTrustProxy();

  stage = "database";
  await connectDatabase();
  await pingDatabase();
  // Do not accept registrations before unique account/token indexes exist.
  await Promise.all([
    mongoose.model("User").init(),
    mongoose.model("PasswordResetToken").init(),

  console.log("MongoDB connection and ping succeeded.");

  stage = "authentication setup";
  const service = createAuthService({
    resetOrigin: process.env.CLIENT_ORIGIN || "http://localhost:5173",
  });
  const auth = createAuth({ connection: mongoose.connection, service });
  await auth.ready();

  stage = "HTTP server";
  server = http.createServer(
    HTTP_LIMITS,
    createApp({
      auth,
      trustProxy,
      // Add your routers here so they sit behind every protection.
      mountRoutes: (app, { auth }) => {
        // app.use("/api/items", itemRoutes(auth));
      },
    }),
  );
  if (process.env.NODE_ENV === "production") {
    server.keepAliveTimeout = PRODUCTION_KEEP_ALIVE_MS;
  }
  server.listen(
    port,
    process.env.NODE_ENV === "production" ? "0.0.0.0" : "127.0.0.1",
    () => {
      console.log(`API listening on port ${port}.`);
    },
  );
  server.on("error", async (error) => {
    console.error(
      error.code === "EADDRINUSE"
        ? "Port is already in use. Choose another PORT in server/.env."
        : "HTTP server failed to start.",
    );
    process.exitCode = 1;
    await shutdown();
  });
} catch (error) {
  // Database errors go through explainDatabaseError so connection strings are never printed.
  console.error(
    `Startup failed during ${stage}:`,
    stage === "database" ? explainDatabaseError(error) : error.message,
  );

}

Step 9 Protect Your Own Routes

server/routes/items.js

Anyone can read, members can write, and only the admin can delete. Pass in auth from server.js through mountRoutes.

import express from "express";
import ItemsController from "../controllers/items.js";

export default function itemRoutes(auth) {

// or changing another account's records.
//
// The most common way people reach someone else's data is by changing an ID in
// a URL, for example /api/demo/reports/<someone-else's-id>. This is called an
// IDOR (insecure direct object reference). These helpers make the safe pattern
// the easy pattern for the routes and controllers you write later:
//
//   const id = readObjectId(req.params.id);
//   const report = await findOwned(Report, { id, ownerId: req.authUser._id });
//   if (!report) throw notFound();
//

// belonging to someone else is never loaded. Both "doesn't exist" and
// "belongs to someone else" return the same 404, so an attacker cannot even
// tell which IDs exist.

export function notFound() {
  return new HttpError(404, "Not found.");

// Accept only a 24-character hexadecimal MongoDB ID string from the URL.
// Anything else (objects, arrays, operators, other formats) is a 404.
export function readObjectId(value) {
  if (typeof value !== "string" || !/^[a-f0-9]{24}$/i.test(value)) {
    throw notFound();
  }
  return value.toLowerCase();
}

// Load a document ONLY if it belongs to ownerId. Returns null otherwise.
export async function findOwned(
  Model,
  { id, ownerId, ownerField = "ownerId", select } = {},
) {
  if (!ownerId) throw new Error("findOwned requires the signed-in owner's ID.");
  const query = Model.findOne({ _id: readObjectId(String(id)), [ownerField]: ownerId });
  if (select) query.select(select);
  return query;
}

// For records already loaded some other way: throw 404 unless owned.
export function assertOwner(document, ownerId, ownerField = "ownerId") {
  if (!document || String(document[ownerField]) !== String(ownerId)) {
    throw notFound();
  }
  return document;
}

Every call to the server goes through this helper. It sends the cookie, adds the CSRF token to each POST, PATCH, and DELETE, and swaps in the new token after login.

// One helper for every call to the backend.
// It handles the two things that made Postman painful:
//   1. The session cookie: the browser stores and sends it automatically
//      (credentials: "same-origin").
//   2. The CSRF token: fetched once, sent as X-CSRF-Token on every
//      POST/PATCH/DELETE, and swapped for the new one the server returns at login.
// Pages don't call this directly; they use the small files in src/services/.

let csrfToken = null;

// The exact message the server sends when the CSRF token doesn't match the session
// (security/auth.js). Usually it means the session expired or was replaced.
const CSRF_FAILED = "Refresh the form and try again.";

export async function refreshCsrf() {
  const response = await fetch("/api/auth/csrf", { credentials: "same-origin" });
  const data = await response.json();
  csrfToken = data.csrfToken;
  return csrfToken;
}

async function send(method, path, body, options) {
  const headers = {};
  if (body !== undefined) headers["Content-Type"] = "application/json";

  const isWrite = method !== "GET";
  if (isWrite && !options.skipCsrf) {
    if (!csrfToken) await refreshCsrf();
    headers["X-CSRF-Token"] = csrfToken;
  }

  const response = await fetch(path, {
    method,
    headers,
    credentials: "same-origin",
    body: body === undefined ? undefined : JSON.stringify(body),
  });

  // Some replies (204 No Content) have no body, so read text first.
  const text = await response.text();
  let data = null;
  try {
    data = text ? JSON.parse(text) : null;
  } catch {
    data = text;
  }

  // Login hands back a NEW token (the old one stops working). Keep it.
  if (data && typeof data === "object" && data.csrfToken) csrfToken = data.csrfToken;
  // After logout the session is gone, so the token is useless too.
  if (path === "/api/auth/logout" && response.ok) csrfToken = null;

  return { status: response.status, ok: response.ok, data };
}

// api("POST", "/api/games", { title: "..." })  ->  { status, ok, data }
// options.skipCsrf = true sends the request WITHOUT the token (only the route
// tester uses it, to prove the server blocks that). Never use it in real pages.
export async function api(method, path, body, options = {}) {
  const result = await send(method, path, body, options);

  // If the session expired, the old token stops matching. Get a fresh token and
  // try once more. (Only this site's own JavaScript can read the token, so a
  // retry here doesn't help an attacker on another site.)
  const tokenRejected =
    result.status === 403 && result.data && result.data.error === CSRF_FAILED;
  if (tokenRejected && method !== "GET" && !options.skipCsrf) {
    await refreshCsrf();
    return send(method, path, body, options);
  }
  return result;
}

// Shared response handling: return the data when the request worked, otherwise throw an Error
// with the server's message so the page can show it.
import { api } from "../api.js";

export class ApiError extends Error {
  constructor(status, message) {
    super(message);
    this.status = status; // 0 = couldn't reach the server at all
  }
}

// When the server says "not signed in" (401) anywhere except the login form,
// the session has ended (expired, or logged out in another tab). Tell the
// AuthContext so the header stops showing a username.
export const SIGNED_OUT_EVENT = "app:signed-out";

export async function request(method, path, body) {
  let result;
  try {
    result = await api(method, path, body);
  } catch {
    throw new ApiError(0, "Can't reach the server. Check your connection and try again.");
  }

  if (!result.ok) {
    if (result.status === 401 && path !== "/api/auth/login") {
      window.dispatchEvent(new Event(SIGNED_OUT_EVENT));
      throw new ApiError(401, "Your session ended. Log in again to continue.");
    }
    const fromServer = result.data && typeof result.data === "object" ? result.data.error : "";
    throw new ApiError(result.status, fromServer || `Request failed (${result.status}).`);
  }
  return result.data;
}

// A friendly message for errors on forms. Pages pass in their own wording for
// the statuses they care about, e.g. { 401: "Wrong email or password." }.
export function messageFor(error, overrides = {}) {
  if (error instanceof ApiError && overrides[error.status]) return overrides[error.status];
  if (error instanceof ApiError && error.status === 429) {
    return "Too many tries in a short time. Wait a few minutes, then try again.";
  }
  return error.message || "Something went wrong. Try again.";
}

// Who is signed in, shared with every page.
// The server is the real gatekeeper (sessions + requireAuth/requireAdmin); this
// only decides what to SHOW. Nothing secret is kept in the browser: no password,
// no session ID, nothing in localStorage. The session cookie is httpOnly, so
// JavaScript can't read it at all.
import { createContext, useContext, useEffect, useRef, useState } from "react";
import authApi from "../services/authApi.js";
import { SIGNED_OUT_EVENT } from "../services/request.js";

const AuthContext = createContext(null);

export function AuthProvider({ children }) {
  const [user, setUser] = useState(null); // { id, username, role } or null
  const [checking, setChecking] = useState(true); // true until /me answers
  const [sessionEnded, setSessionEnded] = useState(false); // shows the "signed out" banner
  const userRef = useRef(null); // the latest user, for the listener below
  useEffect(() => {
    userRef.current = user;
  }, [user]);

  // On first load, ask the server whether this browser already has a session.
  useEffect(() => {
    let ignore = false;
    authApi
      .me()
      .then((found) => {
        if (!ignore) setUser(found);
      })
      .catch(() => {
        if (!ignore) setUser(null);
      })
      .finally(() => {
        if (!ignore) setChecking(false);
      });
    return () => {
      ignore = true;
    };
  }, []);

  // A request found the session had ended: show the visitor view again.
  useEffect(() => {
    const forget = () => {
      if (userRef.current) setSessionEnded(true);
      setUser(null);
    };
    window.addEventListener(SIGNED_OUT_EVENT, forget);
    return () => window.removeEventListener(SIGNED_OUT_EVENT, forget);
  }, []);

  async function login(fields) {
    const signedIn = await authApi.login(fields);
    setUser(signedIn);
    setSessionEnded(false);
    return signedIn;
  }

  async function logout() {
    await authApi.logout();
    setUser(null);
  }

  const value = {
    user,
    checking,
    isAdmin: user?.role === "admin",
    login,
    logout,
    register: authApi.register,
    sessionEnded,
    dismissSessionEnded: () => setSessionEnded(false),
  };
  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}

  if (username.length < 3 || username.length > 35) problems.username = "Use 3 to 35 characters.";
  else if (!USERNAME_PATTERN.test(username)) problems.username = "Use letters, numbers, _ - or . and start and end with a letter or number.";
  if (!/^\S+@\S+\.\S+$/.test(email)) problems.email = "Enter an email address like name@example.com.";
  if (password.length < 15 || password.length > 128) problems.password = "Use 15 to 128 characters.";
  if (confirm !== password) problems.confirm = "The two passwords don't match.";
  return problems;
}

function LoginForm({ onDone }) {
  const { login } = useAuth();
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [error, setError] = useState("");
  const [pending, setPending] = useState(false);
  const inFlight = useRef(false);

  async function handleSubmit(event) {
    event.preventDefault();
    if (inFlight.current) return;
    inFlight.current = true;
    setPending(true);
    setError("");
    try {
      await login({ email: email.trim(), password });
      onDone();
    } catch (problem) {
      setError(messageFor(problem, { 401: "That email and password don't match an account." }));
      setPassword("");
    } finally {
      inFlight.current = false;
      setPending(false);
    }
  }

  return (
    <form onSubmit={handleSubmit}>
      <h2 className="access__title">Welcome back</h2>
      <fieldset disabled={pending}>
        <div className="field">
          <label htmlFor="login-email">Email</label>
          <input id="login-email" className="input" type="email" autoComplete="email" required value={email} onChange={(event) => setEmail(event.target.value)} />
        </div>
        <div className="field">
          <label htmlFor="login-password">Password</label>
          <PasswordInput id="login-password" autoComplete="current-password" value={password} onChange={setPassword} />

}

function RegisterForm({ onDone, onUseLogin }) {
  const { register, login } = useAuth();
  const [form, setForm] = useState({ username: "", email: "", password: "", confirm: "" });
  const [problems, setProblems] = useState({});
  const [error, setError] = useState("");
  const [pending, setPending] = useState(false);
  const inFlight = useRef(false);

  function update(name, value) {
    setForm((previous) => ({ ...previous, [name]: value }));
    setProblems((previous) => ({ ...previous, [name]: undefined }));
  }

  async function handleSubmit(event) {
    event.preventDefault();
    if (inFlight.current) return;
    const fields = { ...form, username: form.username.trim(), email: form.email.trim() };
    const found = checkRegistration(fields);
    setProblems(found);
    if (Object.keys(found).length) return;

    inFlight.current = true;
    setPending(true);
    setError("");
    try {
      await register({ username: fields.username, email: fields.email, password: fields.password });
      // The server never says whether an email was already taken, so try
      // signing in with what was just typed.
      try {
        await login({ email: fields.email, password: fields.password });
        onDone();
      } catch (signInProblem) {
        setError(
          signInProblem.status === 429
            ? "Too many sign-in attempts from this network. Wait a few minutes, then log in."
            : "Your request went through, but you couldn't be signed in automatically. If this email already has an account, log in with it instead.",
        );
      }
    } catch (problem) {
      setError(messageFor(problem, { 409: "That username isn't available. Try another one." }));
    } finally {
      inFlight.current = false;
      setPending(false);
    }
  }

  const describe = (name, hintId) => [hintId, problems[name] ? `register-${name}-error` : null].filter(Boolean).join(" ") || undefined;

  return (
    <form onSubmit={handleSubmit} noValidate>
      <h2 className="access__title">Create an account</h2>
      <fieldset disabled={pending}>
        <div className="field">
          <label htmlFor="register-username">Username</label>
          <input id="register-username" className="input" autoComplete="username" value={form.username} onChange={(event) => update("username", event.target.value)} aria-invalid={Boolean(problems.username)} aria-describedby={describe("username", "register-username-hint")} />
          <p id="register-username-hint" className="field__hint">3 to 35 letters or numbers. You can also use _ - and .</p>
          {problems.username && <p id="register-username-error" className="field__error">{problems.username}</p>}
        </div>
        <div className="field">
          <label htmlFor="register-email">Email</label>
          <input id="register-email" className="input" type="email" autoComplete="email" value={form.email} onChange={(event) => update("email", event.target.value)} aria-invalid={Boolean(problems.email)} aria-describedby={describe("email")} />
          {problems.email && <p id="register-email-error" className="field__error">{problems.email}</p>}
        </div>
        <div className="field">
          <label htmlFor="register-password">Password</label>
          <PasswordInput id="register-password" autoComplete="new-password" value={form.password} onChange={(value) => update("password", value)} invalid={Boolean(problems.password)} describedBy={describe("password", "register-password-hint")} />
          <p id="register-password-hint" className="field__hint">At least 15 characters. A short sentence you'll remember works well.</p>
          {problems.password && <p id="register-password-error" className="field__error">{problems.password}</p>}
        </div>
        <div className="field">
          <label htmlFor="register-confirm">Type the password again</label>
          <PasswordInput id="register-confirm" autoComplete="new-password" value={form.confirm} onChange={(value) => update("confirm", value)} invalid={Boolean(problems.confirm)} describedBy={describe("confirm")} />
          {problems.confirm && <p id="register-confirm-error" className="field__error">{problems.confirm}</p>}
        </div>
        {error && (
          <p className="form-error" role="alert">
            {error}{" "}
            {error.includes("log in") && <button type="button" className="text-button" onClick={onUseLogin}>Go to Log in</button>}
          </p>
        )}
        <div className="form-actions">
          <button type="submit" className="button button--primary access__submit">{pending ? "Creating account..." : "Create account"}</button>
        </div>
      </fieldset>
    </form>
  );
}

// Password box with a Show/Hide button (handy for 15+ character passwords).
export function PasswordInput({ id, value, onChange, autoComplete, invalid = false, describedBy }) {
  const [visible, setVisible] = useState(false);
  return (
    <div className="password">

        className="input"

}

export default function AdminLogin() {
  usePageTitle("Owner sign-in");
  const { user, checking, login } = useAuth();
  const navigate = useNavigate();
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [code, setCode] = useState("");
  const [error, setError] = useState("");
  const [pending, setPending] = useState(false);
  const inFlight = useRef(false);

    event.preventDefault();
    if (inFlight.current) return;
    const digits = code.replace(/\s/g, "");

      setError("Enter the 6-digit code from your authenticator app.");
      return;
    }
    inFlight.current = true;
    setPending(true);
    setError("");

      const signedIn = await login({ email: email.trim(), password, code: digits });

    } catch (problem) {
      setError(
        messageFor(problem, {
          401: "That didn't work. Check your email and password, then wait for a new code and try again (each code works once).",
        }),
      );
      setPassword("");
      setCode("");
    } finally {
      inFlight.current = false;
      setPending(false);
    }
  }

    <div className="page page--narrow">
      <form className="panel owner-login" onSubmit={handleSubmit} noValidate>

        <p className="owner-login__intro">For the site owner. You'll need the 6-digit code from your authenticator app.</p>

            <label htmlFor="owner-email">Email</label>
            <input id="owner-email" className="input" type="email" autoComplete="username" value={email} onChange={(event) => setEmail(event.target.value)} />
          </div>
          <div className="field">
            <label htmlFor="owner-password">Password</label>
            <PasswordInput id="owner-password" autoComplete="current-password" value={password} onChange={setPassword} />

              autoComplete="one-time-code"
              pattern="\d{6}"

          </div>

            <button type="submit" className="button button--primary">{pending ? "Checking..." : "Sign in"}</button>
          </div>
        </fieldset>
      </form>
    </div>
  );
}

Before Going Live

  • Set NODE_ENV=production and TRUST_PROXY_HOPS on the host. The server will not start without them.
  • Set a real SESSION_SECRET and ADMIN_MFA_KEY in the host's settings. Never reuse the local ones.
  • Set CLIENT_ORIGIN to your https address with no slash on the end.
  • Use a database user with only readWrite on your project's database.
  • Add your username to RESERVED_USERNAMES.
  • Run npm audit, and confirm .env is not in Git.
  • Rate limits in app.js and auth.js live in memory, so they reset on restart and count per server. Stop large floods before they reach the server with a service like Cloudflare.
  • Password reset stays off until you connect an email service.
  • Keep a database backup. These layers make attacks harder, but no site is impossible to hack.
